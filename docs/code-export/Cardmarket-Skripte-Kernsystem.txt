# Resume Cardmarket account review on the dedicated local Chrome profile.
# Authentication and login checks remain entirely in Chrome; no cookie export.
param([switch]$ResumeAfterClearance,[switch]$ManualResume)
$ErrorActionPreference="Stop"
$base=Join-Path $env:LOCALAPPDATA "Cardyssey\CardmarketCollector"
$profile=Join-Path $env:LOCALAPPDATA "Cardyssey\CardmarketEdgeProfile"
$runner=Join-Path $base "Cardmarket-Queue-Runner.ps1"
New-Item -ItemType Directory -Force -Path $base | Out-Null
if (!(Test-Path $runner)) {throw "Cardmarket queue runner missing"}
$already=Get-CimInstance Win32_Process | Where-Object {$_.CommandLine -match "-File.*Cardmarket-Queue-Runner[.]ps1" -and $_.Name -eq "powershell.exe"}
if ($already) {return}
# Durable provider-specific cooldown across Windows Task Scheduler launches.
# Never assume that a persistent login or cf_clearance guarantees access.
# Manual resume is still allowed only after the same live login/Block check.
if (-not $ManualResume){
  foreach($lang in @('de','en')){
    $progressFile=Join-Path $base ("progress-$lang.json")
    if(-not (Test-Path $progressFile)){continue}
    try{
      $x=Get-Content -Raw -Encoding UTF8 $progressFile | ConvertFrom-Json
      $last=[datetime]::Parse([string]$x.updated_at).ToUniversalTime()
      $cooldown= switch([string]$x.status){
        'BLOCKED' {240;break}
        'PROVIDER_TIMEOUT' {90;break}
        'CONNECTION_LOST' {60;break}
        'LOGIN_REQUIRED' {20;break}
        'LOGIN_UNVERIFIED' {20;break}
        'BATCH_COMPLETE_MORE_PENDING' {45;break}
        'COMPLETED_WITH_DEFERRED' {90;break}
        default {0}
      }
      if($cooldown -gt 0 -and [DateTime]::UtcNow -lt $last.AddMinutes($cooldown)){
        Write-Output ('CARDMARKET_COOLDOWN '+$lang+' '+$x.status+' until '+$last.AddMinutes($cooldown).ToString('o'))
        return
      }
    }catch{
      Write-Output ('CARDMARKET_PROGRESS_UNREADABLE '+$lang)
      return
    }
  }
}

$ready=$false
try {$null=Invoke-RestMethod -TimeoutSec 2 "http://127.0.0.1:9223/json/list";$ready=$true} catch {}
if (!$ready) {
  $existing=Get-CimInstance Win32_Process | Where-Object {$_.Name -eq "msedge.exe" -and $_.CommandLine -match "CardmarketEdgeProfile" -and $_.CommandLine -notmatch "--type="} | Select-Object -First 1
  if ($existing) {
    $p=Get-Process -Id $existing.ProcessId -ErrorAction SilentlyContinue
    if ($p) {$null=$p.CloseMainWindow();Start-Sleep -Seconds 3}
  }
  $chrome="C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe"
  Start-Process $chrome -ArgumentList @("--user-data-dir=$profile", "--remote-debugging-port=9223", "--remote-allow-origins=http://localhost", "--no-first-run", "https://www.cardmarket.com/de/Pokemon")
  for($i=0;$i -lt 25;$i++) {
    Start-Sleep -Seconds 1
    try {$null=Invoke-RestMethod -TimeoutSec 2 "http://127.0.0.1:9223/json/list";$ready=$true;break} catch {}
  }
}
if (!$ready) {throw "Cardmarket browser debugging endpoint unavailable; preserve profile and retry later"}
# Both automatic and manual resume obey the same provider/login gate.
$diagnose=Join-Path $base 'Cardmarket-Session-Status.ps1'
if(Test-Path $diagnose){
  $health=& $diagnose | ConvertFrom-Json
  $progress=Join-Path $base 'progress-de.json'
  if($health.status -ne 'AVAILABLE' -or $health.login -ne 'AUTHENTICATED'){
    $pause=if($health.status -eq 'BLOCKED'){'BLOCKED'}elseif($health.login -eq 'LOGIN_REQUIRED'){'LOGIN_REQUIRED'}else{'LOGIN_UNVERIFIED'}
    if(Test-Path $progress){
      $x=Get-Content -Raw $progress | ConvertFrom-Json
      $x.status=$pause;$x.updated_at=[DateTime]::UtcNow.ToString('o')
      $x|ConvertTo-Json -Compress|Set-Content -Encoding UTF8 -Path $progress
    }
    Write-Output ('CARDMARKET_PAUSED '+$pause)
    return
  }
}
Start-Process -FilePath "powershell.exe" -WindowStyle Hidden -ArgumentList @("-NoProfile","-ExecutionPolicy","Bypass","-File",$runner,"-BatchLimit","16","-DelaySeconds","10","-MinRequestGapSeconds","10","-Languages","both","-RunMinutes","40","-ScanMode","identity","-RequireLogin") -RedirectStandardOutput (Join-Path $base "language-queue-stdout.log") -RedirectStandardError (Join-Path $base "language-queue-stderr.log")

# Cardmarket scanning is scheduled on Windows and shares one Edge CDP tab.
# Rotate German/English batches so neither backlog can starve the other.
# Never bypass a provider challenge or queue a retry indefinitely.
param(
  [ValidateRange(1,1000)][int]$BatchLimit = 16,
  [ValidateRange(4,120)][int]$DelaySeconds = 10,
  [ValidateRange(4,90)][int]$MinRequestGapSeconds = 10,
  [ValidateSet('de','en','both')][string]$Languages = 'de',
  [ValidateRange(0,110)][int]$RunMinutes = 40,
  [ValidateSet('identity','missing')][string]$ScanMode='identity',
  [switch]$RequireLogin
)
$ErrorActionPreference='Stop'
$base=Join-Path $env:LOCALAPPDATA 'Cardyssey\CardmarketCollector'
$collector=Join-Path $base 'cardmarket-language-browser-collector.ps1'
if(!(Test-Path $collector)){throw 'Updated language collector missing'}
$selectedLanguages=if($Languages -eq 'both'){@('de','en')}else{@($Languages)}
$done=@{}
foreach($l in $selectedLanguages){$done[$l]=$false}
$deadline=[DateTime]::UtcNow.AddMinutes($RunMinutes)
$stop=$false
do {
  foreach($language in $selectedLanguages){
    if($done[$language]){continue}
    if($RunMinutes -gt 0 -and [DateTime]::UtcNow -ge $deadline){break}
    Write-Output ('LANGUAGE_BEGIN '+$language+' '+[DateTime]::UtcNow.ToString('o'))
    & $collector -Language $language -InputCsv (Join-Path $base "queue-$language.csv") -OutputDir $base -CdpPort 9223 -BatchLimit $BatchLimit -DelaySeconds $DelaySeconds -MinRequestGapSeconds $MinRequestGapSeconds -ScanMode $ScanMode -RequireLogin:$RequireLogin
    if(!$?){throw "Collector failed for $language"}
    $progress=Get-Content -Raw (Join-Path $base "progress-$language.json") | ConvertFrom-Json
    Write-Output ('LANGUAGE_END '+$language+' status='+$progress.status+' processed='+$progress.done)
    if($progress.status -in @('BLOCKED','PROVIDER_TIMEOUT','CONNECTION_LOST','LOGIN_REQUIRED','LOGIN_UNVERIFIED')){
      # A provider challenge/login failure may affect both language sessions.
      # Do not hammer the other queue. The Windows task safely resumes later.
      Write-Output 'STOP_ON_PROVIDER_OR_CONNECTION_ERROR'
      $stop=$true;break
    }
    if($progress.status -in @('COMPLETE','COMPLETED','COMPLETED_WITH_DEFERRED') -or
       [int]$progress.done -ge [int]$progress.total){
      $done[$language]=$true
    } elseif($progress.status -ne 'BATCH_COMPLETE_MORE_PENDING'){
      Write-Output ('UNEXPECTED_PROGRESS_STATE '+$language+' '+$progress.status)
      $stop=$true;break
    }
    if($RunMinutes -gt 0 -and [DateTime]::UtcNow -lt $deadline){
      Start-Sleep -Seconds 12
    }
  }
  if($stop -or $RunMinutes -eq 0 -or [DateTime]::UtcNow -ge $deadline){break}
  $pending=@($selectedLanguages | Where-Object {-not $done[$_]})
  if($pending.Count -eq 0){break}
  Start-Sleep -Seconds 25
}while($true)
Write-Output ('RUNNER_END '+[DateTime]::UtcNow.ToString('o')+
  ' completed='+(@($selectedLanguages | Where-Object {$done[$_]} ) -join ',')+
  ' stopped='+$stop)

# Read-only watch of the existing Cardmarket Edge session.
# Never solve a challenge; resume only when Cardmarket has cleared it itself.
$ErrorActionPreference='Stop'
$base=Join-Path $env:LOCALAPPDATA 'Cardyssey\CardmarketCollector'
$log=Join-Path $base 'resume-watchdog.log'
function Note($value) {
  ('{0} {1}' -f [DateTime]::UtcNow.ToString('o'),$value) | Add-Content -Path $log -Encoding UTF8
}
try {
  # Diagnose the persistent profile on every watchdog pass, even if a scan is
  # running. Read-only: never access stored passwords or bypass the provider.
  $checker=Join-Path $base 'Cardmarket-Session-Status.ps1'
  if(Test-Path $checker){
    $diag=& $checker | ConvertFrom-Json
    if($diag.status -eq 'BLOCKED'){Note 'CHALLENGE_STILL_PRESENT';return}
    if($diag.login -eq 'LOGIN_REQUIRED'){Note 'CARDMARKET_LOGIN_REQUIRED';return}
    if($diag.status -ne 'AVAILABLE' -or $diag.login -ne 'AUTHENTICATED'){Note 'CARDMARKET_SESSION_NOT_VERIFIED';return}
  }
  $busy=@(Get-CimInstance Win32_Process | Where-Object {
     $_.Name -eq 'powershell.exe' -and $_.CommandLine -match '-File.*Cardmarket-Queue-Runner[.]ps1'
  })
  if($busy.Count -gt 0){return}
  $states=@()
  foreach($language in @('de','en')){
    $progressPath=Join-Path $base ("progress-$language.json")
    if(-not (Test-Path $progressPath)){continue}
    $state=Get-Content -Raw -Encoding UTF8 $progressPath | ConvertFrom-Json
    $last=[DateTime]::Parse([string]$state.updated_at).ToUniversalTime()
    $stalled=($state.status -eq 'RUNNING' -and $last -lt [DateTime]::UtcNow.AddMinutes(-6))
    if($stalled){
      $state.status='INTERRUPTED'
      $state.updated_at=[DateTime]::UtcNow.ToString('o')
      $state | ConvertTo-Json -Compress | Set-Content -Encoding UTF8 -Path $progressPath
      $last=[DateTime]::UtcNow
      Note ('STALE_RUNNING_INTERRUPTED '+$language)
    }
    $states+= [pscustomobject]@{language=$language;status=[string]$state.status;last=$last}
  }
  if($states.Count -eq 0){return}
  $remaining=@($states | Where-Object { $_.status -notin @('COMPLETED','COMPLETE') })
  if($remaining.Count -eq 0){return}
  # Do not restart a blocked provider immediately just because a static page
  # or cached logout link can still be read. Only read-only probes are allowed
  # before the full backoff expires. After normal batches use shorter breaks.
  foreach($s in $remaining){
    $minutes=switch($s.status){
      'BLOCKED' {240;break}
      'PROVIDER_TIMEOUT' {90;break}
      'CONNECTION_LOST' {60;break}
      'LOGIN_REQUIRED' {20;break}
      'LOGIN_UNVERIFIED' {20;break}
      'BATCH_COMPLETE_MORE_PENDING' {45;break}
      'COMPLETED_WITH_DEFERRED' {90;break}
      default {0}
    }
    if($minutes -gt 0 -and [DateTime]::UtcNow -lt $s.last.AddMinutes($minutes)){
      return
    }
  }
  $pages=Invoke-RestMethod -Uri 'http://127.0.0.1:9223/json/list' -TimeoutSec 5
  $active=$pages | Where-Object { $_.type -eq 'page' -and ([string]$_.url).StartsWith('https://www.cardmarket.com/') } | Select-Object -First 1
  if(!$active){Note 'NO_CARDMARKET_TAB';return}
  if([string]$active.title -match '(?i)nur einen moment|just a moment|attention required|security check|checking your browser|access denied|cloudflare to restrict access|too many requests'){
    Note 'CHALLENGE_STILL_PRESENT';return
  }
  # Title alone cannot prove a valid product page, but after a prior block
  # it can trigger one cautious attempt. Collector rechecks every response.
  $launcher=Join-Path $base 'Cardmarket-Collector-Launcher.ps1'
  Note 'PAGE_VISIBLE_ATTEMPT_RESUME'
  & $launcher -ResumeAfterClearance
} catch {
  Note ('WATCH_ERROR '+$_.Exception.Message.Substring(0,[Math]::Min(130,$_.Exception.Message.Length)))
}

# Read-only diagnosis of the SAME persistent Cardmarket Edge profile.
# Never accesses password manager contents, credentials, cookies or tokens.
$ErrorActionPreference='Stop'
$base=Join-Path $env:LOCALAPPDATA 'Cardyssey\CardmarketCollector'
$dest=Join-Path $base 'browser-session-status.json'
$state=[ordered]@{checkedAt=[DateTime]::UtcNow.ToString('o');provider='CARDMARKET';status='UNKNOWN';reason='';login='UNKNOWN';profileExists=(Test-Path (Join-Path $env:LOCALAPPDATA 'Cardyssey\CardmarketEdgeProfile'))}
$ws=$null
try {
  $targets=Invoke-RestMethod -Uri 'http://127.0.0.1:9223/json/list' -TimeoutSec 4
  $page=$targets | Where-Object {$_.type -eq 'page' -and ([string]$_.url) -match '^https://www[.]cardmarket[.]com/'} | Select-Object -First 1
  if(!$page){$state.status='NO_CARDMARKET_TAB';$state.reason='Browser tab unavailable'}
  else {
    if([string]$page.title -match '(?i)nur einen moment|just a moment|attention required|security check|checking your browser|access denied|cloudflare to restrict access|too many requests'){
      $state.status='BLOCKED';$state.reason='CLOUDFLARE_CHALLENGE'
    } elseif (!$page.webSocketDebuggerUrl){
      $state.status='BROWSER_UNAVAILABLE'
    } else {
      $ws=[Net.WebSockets.ClientWebSocket]::new()
      $ws.Options.SetRequestHeader('Origin','http://localhost')
      $cts=[Threading.CancellationTokenSource]::new(5000)
      try{$ws.ConnectAsync([uri]$page.webSocketDebuggerUrl,$cts.Token).GetAwaiter().GetResult()|Out-Null}finally{$cts.Dispose()}
      $js='JSON.stringify({title:document.title,blocked:/(nur einen moment|just a moment|attention required|checking your browser|captcha|access denied|cloudflare to restrict access|too many requests)/i.test(document.title)||/(?:you have been blocked|you are unable to access|access denied|error 1020|too many requests|temporarily rate.limited)/i.test((document.body?.innerText||"").slice(0,1700)),loginForm:!!document.querySelector("form[action*=Login],form[action*=login],input[type=password]"),logoutLink:!!document.querySelector("a[href*=User_Logout],form[action*=User_Logout]"),accountHint:!!document.querySelector("a[href*=User_Profile],a[href*=User_Account],a[href*=MyAccount]"),pageReady:document.readyState})'
      $packet=@{id=7;method='Runtime.evaluate';params=@{expression=$js;returnByValue=$true}}|ConvertTo-Json -Compress -Depth 5
      $buffer=[Text.Encoding]::UTF8.GetBytes($packet)
      $ws.SendAsync([ArraySegment[byte]]::new($buffer),[Net.WebSockets.WebSocketMessageType]::Text,$true,[Threading.CancellationToken]::None).GetAwaiter().GetResult()|Out-Null
      $result=$null
      $deadline=[DateTime]::UtcNow.AddSeconds(5)
      while([DateTime]::UtcNow -lt $deadline){
        $bytes=New-Object byte[] 65536
        $wait=[Threading.CancellationTokenSource]::new(1500)
        try{$answer=$ws.ReceiveAsync([ArraySegment[byte]]::new($bytes),$wait.Token).GetAwaiter().GetResult()}catch{break}finally{$wait.Dispose()}
        if(!$answer.EndOfMessage){continue}
        $frame=[Text.Encoding]::UTF8.GetString($bytes,0,$answer.Count)|ConvertFrom-Json
        if($frame.id -ne 7){continue}
        if($frame.result.result.value){$result=$frame.result.result.value | ConvertFrom-Json}
        break
      }
      if(!$result){$state.status='BROWSER_STATE_UNREADABLE'}
      elseif($result.blocked){$state.status='BLOCKED';$state.reason='CLOUDFLARE_CHALLENGE'}
      elseif($result.logoutLink){$state.status='AVAILABLE';$state.login='AUTHENTICATED';$state.reason='VISIBLE_LOGOUT_LINK'}
      elseif($result.loginForm){$state.status='AVAILABLE';$state.login='LOGIN_REQUIRED';$state.reason='VISIBLE_LOGIN_FORM'}
      else{$state.status='AVAILABLE';$state.login='UNVERIFIED';$state.reason='NO_RELIABLE_LOGIN_INDICATOR'}
    }
  }
} catch {$state.status='BROWSER_UNAVAILABLE';$state.reason='LOCAL_BROWSER_CONNECTION_ERROR'}
finally{if($ws){try{$ws.Dispose()}catch{}}}
New-Item -ItemType Directory -Force -Path $base|Out-Null
($state|ConvertTo-Json -Compress -Depth 3)|Set-Content -Encoding UTF8 -Path $dest
$state|ConvertTo-Json -Compress -Depth 3

# --- CDP-Empfang: nativer Pump-Thread (kein PowerShell-Runspace noetig) --------
# Der ReceiveAsync wird NIE abgebrochen (keine Timeout-Cancellation), damit der
# WebSocket nie in "Aborted" faellt. Gesamtfristen regelt Send-Cdp ueber die Queue.
if (-not ('CdpPump' -as [type])) {
  Add-Type -TypeDefinition @"
using System;
using System.Net.WebSockets;
using System.Text;
using System.Collections.Concurrent;
using System.Threading;
public static class CdpPump {
  public static void Start(ClientWebSocket ws, ConcurrentQueue<string> queue,
                            SemaphoreSlim avail, CancellationToken token) {
    var t = new Thread(() => {
      var buf = new byte[1048576];
      var ms  = new System.IO.MemoryStream();
      try {
        while (!token.IsCancellationRequested) {
          var res = ws.ReceiveAsync(new ArraySegment<byte>(buf), token).GetAwaiter().GetResult();
          ms.Write(buf, 0, res.Count);
          if (!res.EndOfMessage) continue;
          queue.Enqueue(Encoding.UTF8.GetString(ms.ToArray()));
          ms.SetLength(0);
          avail.Release();
        }
      } catch (Exception e) {
        queue.Enqueue("STREAM_DEAD|" + e.Message);
        try { avail.Release(); } catch {}
      }
    });
    t.IsBackground = true;
    t.Start();
  }
}
"@
}

$script:ws      = $null
$script:rxQueue = $null
$script:rxAvail = $null
$script:rxCts   = $null

function Connect-Cdp {
  # Alte Verbindung ordentlich beenden, falls vorhanden.
  if ($script:ws) {
    try { $script:rxCts.Cancel() } catch {}
    try { $script:ws.Dispose() } catch {}
  }
  $targets = (Invoke-RestMethod "http://127.0.0.1:$CdpPort/json/list")
  $t = $targets | Where-Object { $_.type -eq "page" -and $_.url -match "cardmarket.com" } | Select-Object -First 1
  if (!$t) { throw "NO_CDM_TARGET: signierte Cardmarket-Seite im isolierten Chrome-Profil oeffnen." }
  $script:ws = [Net.WebSockets.ClientWebSocket]::new()
  $script:ws.Options.SetRequestHeader("Origin", "http://localhost")
  $script:ws.ConnectAsync([uri]$t.webSocketDebuggerUrl, [threading.cancellationtoken]::None).GetAwaiter().GetResult() | Out-Null
  # Frische Queue + Pump fuer DIESE Socket-Instanz.
  $script:rxQueue = [System.Collections.Concurrent.ConcurrentQueue[string]]::new()
  $script:rxAvail = [System.Threading.SemaphoreSlim]::new(0, [int]::MaxValue)
  $script:rxCts   = [System.Threading.CancellationTokenSource]::new()
  [CdpPump]::Start($script:ws, $script:rxQueue, $script:rxAvail, $script:rxCts.Token)
  $script:cdpRequestId = 0
}
Connect-Cdp

function Send-Cdp([string]$method, [hashtable]$parameters, [int]$TimeoutSec = 20) {
  $script:cdpRequestId += 1
  $currentId = [int]$script:cdpRequestId
  Write-Host ('CDP_SEND '+$method+' id='+$currentId)
  $packet = @{id=$currentId;method=$method;params=$parameters} | ConvertTo-Json -Depth 15 -Compress
  $bytes = [text.encoding]::UTF8.GetBytes($packet)
  $ws.SendAsync([ArraySegment[byte]]::new($bytes), [Net.WebSockets.WebSocketMessageType]::Text, $true,
                [threading.CancellationToken]::None).GetAwaiter().GetResult() | Out-Null
  $deadline = [DateTime]::UtcNow.AddSeconds($TimeoutSec)
  while ([DateTime]::UtcNow -lt $deadline) {
    if (-not $script:rxAvail.Wait($deadline - [DateTime]::UtcNow)) { break }
    $frame = $null
    if (-not $script:rxQueue.TryDequeue([ref]$frame)) { continue }
    if ($frame -like 'STREAM_DEAD|*') { throw ('CDP_STREAM_DEAD: ' + $frame.Substring(12)) }
    $parsed = $frame | ConvertFrom-Json
    if ($parsed.id -ne $currentId) { continue }
    if ($parsed.error) { throw "Browser protocol error: $($parsed.error.message)" }
    Write-Host ('CDP_OK '+$method+' id='+$currentId)
    return $parsed
  }
  Write-Host ('CDP_NO_RESPONSE '+$method+' id='+$currentId+' ws='+$script:ws.State)
  throw 'CDP_TIMEOUT'   # Nur die Frist laeuft ab; der Socket bleibt unangetastet.
}

function Parse-Cardmarket-Euro([string]$text) {
  if ([string]::IsNullOrWhiteSpace($text)) { return $null }
  $m=[regex]::Match($text,'(?<!\d)(?:\d{1,3}(?:[., ]\d{3})+|\d+)(?:[.,]\d{1,2})?(?!\d)')
  if (!$m.Success) {return $null}
  $amount=$m.Value.Trim().Replace(' ','')
  $comma=$amount.LastIndexOf(',')
  $dot=$amount.LastIndexOf('.')
  if($comma -ge 0 -and $dot -ge 0) {
    if($comma -gt $dot){$amount=$amount.Replace('.','').Replace(',','.')}
    else {$amount=$amount.Replace(',','')}
  } elseif($comma -ge 0) {
    if($amount -match ',\d{1,2}$'){$amount=$amount.Replace(',','.')}
    else {$amount=$amount.Replace(',','')}
  } elseif($dot -ge 0 -and $amount -match '^\d{1,3}(?:\.\d{3})+$') {
    $amount=$amount.Replace('.','')
  }
  $number=0.0
  if([double]::TryParse($amount,[Globalization.NumberStyles]::AllowDecimalPoint,[Globalization.CultureInfo]::InvariantCulture,[ref]$number) -and $number -gt 0){return $number}
  return $null
}

function Js([string]$expression, [int]$TimeoutSec = 20) {
  $reply = Send-Cdp 'Runtime.evaluate' @{expression=$expression;returnByValue=$true;awaitPromise=$true} $TimeoutSec
  if ($reply.result.exceptionDetails) { throw 'Browser script exception' }
  return $reply.result.result.value
}

function Is-CardmarketChallenge([string]$title, [bool]$challenge) {
  return ($challenge -or $title -match '(?i)nur einen moment|just a moment|attention required|security check|checking your browser|access denied|cloudflare to restrict access|too many requests')
}

$script:lastProviderRequestUtc=[DateTime]::MinValue
function Wait-CardmarketRequestBudget {
  # Global to this single browser tab/collector process. Limit provider
  # navigation AND AJAX page requests, not merely completed card checks.
  # Keep delays fixed and conservative; never forge browser fingerprints.
  if($script:lastProviderRequestUtc -ne [DateTime]::MinValue){
    $elapsed=([DateTime]::UtcNow-$script:lastProviderRequestUtc).TotalSeconds
    if($elapsed -lt $MinRequestGapSeconds){
      $waitMs=[int][Math]::Ceiling(($MinRequestGapSeconds-$elapsed)*1000)
      if($waitMs -gt 0){Start-Sleep -Milliseconds $waitMs}
    }
  }
  $script:lastProviderRequestUtc=[DateTime]::UtcNow
}
function Assert-CardmarketAuthenticated {
  # An auth indicator can vanish behind a Cloudflare block. Report BLOCKED
  # rather than telling the owner to login again unnecessarily.
  $read=Js 'JSON.stringify({title:document.title,challenge:!!document.querySelector("#challenge-running,#cf-challenge-running")||/(?:you have been blocked|you are unable to access|access denied|error 1020|too many requests)/i.test((document.body?.innerText||"").slice(0,1700)),loggedIn:!!document.querySelector("a[href*=User_Logout],form[action*=User_Logout]")})' | ConvertFrom-Json
  if(Is-CardmarketChallenge ([string]$read.title) ([bool]$read.challenge)){
    throw 'CARDMARKET_PROVIDER_CHALLENGE'
  }
  if($read.loggedIn){return}
  throw 'CARDMARKET_LOGIN_REQUIRED'
}
function Move-To([string]$url) {
  Wait-CardmarketRequestBudget
  Send-Cdp 'Page.navigate' @{url=$url} | Out-Null
  $desired = [uri]$url
  $attemptStart=[DateTime]::UtcNow
  $challengeStart=$null
  # The same persistent browser profile retains lawful session data.
  # Only WAIT for Cardmarket to finish its own challenge; do not click,
  # forge clearance cookies, replay private tokens, or bypass a challenge.
  for ($i = 0; $i -lt 380; $i++) {
    Start-Sleep -Milliseconds 400
    $state = Js 'JSON.stringify({path:location.pathname,href:location.href,ready:document.readyState,body:document.body?.innerText.length||0,title:document.title,challenge:!!document.querySelector("#challenge-running,#cf-challenge-running")||/(?:you have been blocked|you are unable to access|access denied|error 1020|too many requests|temporarily rate.limited)/i.test((document.body?.innerText||"").slice(0,1700))})' 10
    if (!$state) { continue }
    $s = $state | ConvertFrom-Json
    if (Is-CardmarketChallenge ([string]$s.title) ([bool]$s.challenge)) {
      if($null -eq $challengeStart){
        $challengeStart=[DateTime]::UtcNow
        Write-Host 'CARDMARKET_CHALLENGE_WAITING: keeping persistent Edge session open'
      }
      if (([DateTime]::UtcNow-$challengeStart).TotalSeconds -ge $ChallengeWaitSeconds) {
        throw 'CARDMARKET_PROVIDER_CHALLENGE'
      }
      continue
    }
    $actual = [uri]$s.href
    $expectedQuery = [uri]::UnescapeDataString($desired.Query.TrimStart('?').Replace('+',' '))
    $actualQuery   = [uri]::UnescapeDataString($actual.Query.TrimStart('?').Replace('+',' '))
    $sameQuery = ($expectedQuery -eq $actualQuery) -or ($desired.AbsolutePath -ne ($LangCfg.PathPrefix + '/Products/Search'))
    if ($s.path -eq $desired.AbsolutePath -and $sameQuery -and $s.ready -eq 'complete' -and $s.body -gt 300) {
      if ($null -ne $challengeStart) { Write-Host 'CARDMARKET_CHALLENGE_CLEARED: continuing current card' }
      return $true
    }
    if (([DateTime]::UtcNow-$attemptStart).TotalSeconds -gt 45 -and $null -eq $challengeStart){return $false}
  }
  return $false
}

function Append-Observation([string]$path, [string]$line) {
  # WSL admin monitor + other processes can momentarily hold Windows JSONL
  # files without FileShare.Write. Never abort a 30k-case scan for one lock.
  $text=$line + [Environment]::NewLine
  $bytes=[Text.Encoding]::UTF8.GetBytes($text)
  for($attempt=0;$attempt -lt 30;$attempt++) {
    $stream=$null
    try {
      $stream=[IO.FileStream]::new($path,[IO.FileMode]::Append,[IO.FileAccess]::Write,[IO.FileShare]::ReadWrite)
      $stream.Write($bytes,0,$bytes.Length)
      $stream.Flush()
      return
    } catch [IO.IOException] {
      if($attempt -eq 29){throw 'CARDMARKET_OBSERVATION_FILE_LOCK_TIMEOUT'}
      Start-Sleep -Milliseconds (150+100*$attempt)
    } finally {if($stream){$stream.Dispose()}}
  }
}

            $familyCandidate=$matches[0]
            $out.version_family_candidate_id=[string]$familyCandidate.productId
            $out.version_family_candidate_url=[string]$familyCandidate.productUrl
            $out.version_family_status='BROWSER_OBSERVED_PRODUCT_CANDIDATE_PRINT_UNVERIFIED'
          }
        }
      }
      # Trusted queue mapping: skip unreliable text search, but validate the
      # actual idProduct on the destination page before accepting it.
      if ([string]$case.routing_state -eq 'VERIFIED_PRODUCT_ID' -and
          @($allowedIds | Where-Object { $_ -match '^[0-9]+$' }).Count -eq 1 -and
          -not [string]::IsNullOrWhiteSpace([string]$case.known_url)) {
        try {
          $candidate = [uri](([string]$case.known_url).Split('?')[0])
          if ($candidate.Scheme -eq 'https' -and $candidate.Host -eq 'www.cardmarket.com' -and
              $candidate.AbsolutePath.StartsWith($LangCfg.PathPrefix + '/Products/Singles/')) {
            $savedUrl = $candidate.AbsoluteUri
          }
        } catch {}
      }

      # Human-supplied CARD-page hints only: navigate directly to the
      # correct Cardmarket page, but NEVER interpret this as proof of a rare
      # stamp/foil, nor as a verified edition-specific price.
      if(!$savedUrl -and [string]$case.reference_print_verified -eq 'NO' -and
         -not [string]::IsNullOrWhiteSpace([string]$case.reference_product_url)) {
        try {
          $referenceUri=[uri]([string]$case.reference_product_url)
          $guard=[string]$setSlugGuards[[string]$case.set_id]
          if($referenceUri.Scheme -eq 'https' -and $referenceUri.Host -eq 'www.cardmarket.com' -and
             $guard -and $referenceUri.AbsolutePath.StartsWith($LangCfg.PathPrefix + '/Products/Singles/' + $guard + '/')) {
            $savedUrl=$referenceUri.AbsoluteUri
            $out.reference_page_used=$true
            $out.reference_page_evidence=[string]$case.reference_evidence
          }
        } catch {}
      }
      # On special prints, use an independently browser-observed product
      # candidate when no manually trusted page or configured hint exists.
      # Exact print mapping remains UNVERIFIED even if ID + number are true.
      if(!$savedUrl -and $familyCandidate){
        $savedUrl=[string]$familyCandidate.productUrl
        $out.version_family_candidate_used=$true
      }
      if (!$savedUrl -and $latest.ContainsKey($key)) {
        $prior = $latest[$key]
        if (([string]$prior.product_id -in $allowedIds) -and $prior.identity_confirmed -eq $true) {
          try {
            $candidate = [uri](([string]$prior.url).Split('?')[0])
            if ($candidate.Scheme -eq 'https' -and $candidate.Host -eq 'www.cardmarket.com' -and
                $candidate.AbsolutePath.StartsWith($LangCfg.PathPrefix + '/Products/Singles/')) { $savedUrl = $candidate.AbsoluteUri }
          } catch {}
        }
      }
      if (!$savedUrl) {
        $reuse = @($allowedIds | Where-Object { $_ -and $knownProductPages.ContainsKey("$($case.cardyssey_karte)|$_") })
        if ($reuse.Count -eq 1) { $savedUrl = $knownProductPages["$($case.cardyssey_karte)|$($reuse[0])"] }
      }

      # Catalog idProduct is a much stronger lookup key than translated
      # search text. The official feed provides both the product ID and its
      # expansion; we still require the live page to reveal the SAME ID and
      # a matching printed collector number. Never infer the exact print.
      # If Cardmarket no longer supports the direct-ID redirect, disable this
      # optimization for the rest of the batch, and use old text search.
      if(!$savedUrl -and !$catalogIdLookupUnsupported -and
         [string]$case.catalog_singleton_id -match '^[0-9]{4,12}$' -and
         [string]$case.catalog_expansion_id -match '^[0-9]+$' -and
         [string]$case.candidate_status -ne 'SET_EXPANSION_AMBIGUOUS') {
        $directId=[string]$case.catalog_singleton_id
        $lookup='https://www.cardmarket.com/Pokemon/Products?idProduct='+$directId
        $out.catalog_direct_id_attempted=$true
        $resolved=''
        $challenged=$false
        try {
          Wait-CardmarketRequestBudget
          Send-Cdp 'Page.navigate' @{url=$lookup} 12 | Out-Null
          for($attempt=0;$attempt -lt 18;$attempt++) {
            Start-Sleep -Milliseconds 300
            $state=Js 'JSON.stringify({url:location.href,title:document.title,id:document.querySelector("input[name=idProduct]")?.value||"",ready:document.readyState,challenge:!!document.querySelector("#challenge-running,#cf-challenge-running")||/(?:you have been blocked|you are unable to access|access denied|error 1020|too many requests|temporarily rate.limited)/i.test((document.body?.innerText||"").slice(0,1700))})' 8 | ConvertFrom-Json
            if(Is-CardmarketChallenge ([string]$state.title) ([bool]$state.challenge)) {
              $challenged=$true
              break
            }
            if($state.ready -ne 'complete' -or [string]$state.id -ne $directId){continue}
            $landing=[uri]([string]$state.url)
            if($landing.Host -ne 'www.cardmarket.com' -or
               $landing.AbsolutePath -notmatch '^/(de|en)/Pokemon/Products/Singles/') {continue}
            $lastNumber=[regex]::Match($landing.AbsolutePath,'(?i)-[A-Z]{1,8}([0-9]{1,4})$')
            $sameNumber=$false
            if($lastNumber.Success){
              try{$sameNumber=([int]$lastNumber.Groups[1].Value -eq [int]([string]$case.kartennummer))}catch{}
            }
            # A verified set slug overrides product-name similarity.
            $guard=[string]$setSlugGuards[[string]$case.set_id]
            $sameSet=(!$guard -or $landing.AbsolutePath -match ('/Products/Singles/'+[regex]::Escape($guard)+'/'))
            if($sameNumber -and $sameSet){
              # Cardmarket may append the viewer account's filters. NEVER
              # reuse language/condition/reverse flags from a redirect.
              $resolved=$landing.GetLeftPart([UriPartial]::Path)
              break
            }
          }
          if($challenged){throw 'CARDMARKET_PROVIDER_CHALLENGE'}
        } catch {
          if([string]$_ -match 'CARDMARKET_PROVIDER_CHALLENGE'){throw}
          $out.catalog_direct_id_error='DIRECT_ID_LOOKUP_TIMEOUT_OR_UNAVAILABLE'
        }
        if($resolved){
          $savedUrl=$resolved
          $out.catalog_direct_id_verified=$true
          $out.catalog_direct_id_product=$directId
        } else {
          $out.catalog_direct_id_error='NO_VERIFIED_REDIRECT_FALLBACK_TO_SEARCH'
          $catalogIdLookupUnsupported=$true
        }
      }
      if ($savedUrl) {
        $out.search_hits = 1
        $likely = @([pscustomobject]@{url=$savedUrl;text="$name $number"})
      } else {
        # Prefer the official catalog's product name, NOT the translated
        # Cardyssey name. Cardmarket's search rarely indexes localized card
        # names plus collector numbers in the same query.
        $catalogName=[string]$case.catalog_search_name
        $searchQueries = @($out.query)
        if (![string]::IsNullOrWhiteSpace($catalogName) -and
            [string]$case.catalog_singleton_id -match '^[0-9]+$') {
          $searchQueries=@($catalogName,$out.query)
        }
        $likely=@()
        foreach($queryText in $searchQueries){
          $searchPath = $LangCfg.PathPrefix + '/Products/Search'
          $encoded = [uri]::EscapeDataString([string]$queryText)
          $searchUrl = 'https://www.cardmarket.com' + $searchPath + '?searchString=' + $encoded
          if ($searchCache.ContainsKey($searchUrl)) {
            $links=@($searchCache[$searchUrl]);$out.search_cache_hit=$true;$searchCacheHits++
          } else {
            $searchCacheMisses++
            if (!(Move-To $searchUrl)) {
              $out.status='DEFERRED_PROVIDER_TIMEOUT'
              $out.error='SEARCH_PAGE_TIMEOUT'
              Append-Observation $resultsPath ($out | ConvertTo-Json -Depth 6 -Compress)
              $consecutiveProviderTimeouts++
              Save-Progress $processed.Count $runTotal 'RUNNING' $case.cardyssey_karte
              if ($consecutiveProviderTimeouts -ge 3) {
                Save-Progress $processed.Count $runTotal 'PROVIDER_TIMEOUT' $case.cardyssey_karte
                break
              }
              Start-Sleep -Seconds ([Math]::Max(10,$DelaySeconds))
              continue
            }
            if ($RequireLogin -and -not (Js '!!document.querySelector("a[href*=User_Logout],form[action*=User_Logout]")')) {
              throw 'CARDMARKET_LOGIN_REQUIRED'
            }
            $searchExpr = 'JSON.stringify({title:document.title,blocked:/(?:nur einen moment|just a moment|attention required|checking your browser|captcha|robot check|access denied|zugriff verweigert|too many requests)/i.test(document.title+" "+(document.body?.innerText||"").slice(0,1600)),links:[...document.querySelectorAll("a[href*=\"/Products/Singles/\"]")].map(a=>({text:a.textContent.trim().slice(0,140),url:new URL(a.getAttribute("href"),location.origin).href})).filter(x=>x.text)})'
            # Cardmarket's result list is injected AFTER document.readyState
            # becomes complete. Reading immediately caused thousands of false
            # NO_EXACT_NUMBER_SEARCH_HIT records with search_hits=0.
            $searchResult=$null
            for($loadTry=0;$loadTry -lt 18;$loadTry++) {
              $searchResult = (Js $searchExpr 12) | ConvertFrom-Json
              if($searchResult.blocked -or @($searchResult.links).Count -gt 0) { break }
              Start-Sleep -Milliseconds 400
            }
            if ($searchResult.blocked) {
              $out.status='BLOCKED'
              $out.error='CARDMARKET_PROVIDER_CHALLENGE'
              Save-Progress $processed.Count $runTotal 'BLOCKED' $case.cardyssey_karte
              break
            }
            $consecutiveProviderTimeouts=0
            $links = @($searchResult.links)
            if($links.Count -eq 0){
              $out.search_results_unverified_empty=$true
              # Keep an empty search result as unverified; retry later.
              # Previously 8,902 cards were incorrectly recorded as missing.
              $out.error='ASYNC_SEARCH_RESULT_EMPTY_OR_NOT_LOADED'
            }
            if ($links.Count -gt 0) {
              if ($searchCache.Count -ge 250) { $searchCache.Clear() }
              $searchCache[$searchUrl] = $links
            }
          }
          $out.search_hits = $links.Count
          $out.query=[string]$queryText
          # Hard gates: exact trailing collector number and independently
          # confirmed expansion slug (when available). Brackets/V2 are not
          # automatically treated as a stamp. This is only PAGE ranking.
          $guard=[string]$setSlugGuards[[string]$case.set_id]
          if(!$guard -and !$isSpecialPrint){$guard=[string]$derivedSetSlugs[[string]$case.set_id]}
          $rank=Resolve-CardmarketSearchCandidates -Links $links -PrintedNumber $number -SetSlug $guard -Name $name -Stamp ([string]$case.stempel) -Foil ([string]$case.foil) -PrintType ([string]$case.drucktyp) -AllowedProductIds $allowedIds
          $out.score_matching_status=[string]$rank.reason
          $out.score_candidate_count=[int]$rank.candidateCount
          $out.score_matched_set=[bool]$guard
          $out.score_top_candidate_urls=@($rank.candidates | Select-Object -First 5 | ForEach-Object {$_.url})
          if($rank.uniquePageCandidate){
            $likely=@([pscustomobject]@{url=$rank.uniquePageCandidate.url;text=$rank.uniquePageCandidate.text})
            # The chosen product must still pass the live idProduct check.
            # NEVER infer the physical foil/stamp or a language-specific price.
            $out.score_page_candidate_only=$true
          } else {
            $likely=@($rank.candidates | Select-Object -First 8 | ForEach-Object {[pscustomobject]@{url=$_.url;text=$_.text}})
          }
          if ($likely.Count -gt 0) { break }
        }
        if ($out.status -eq 'DEFERRED_PROVIDER_TIMEOUT') { continue }
        if ($out.status -eq 'PROVIDER_TIMEOUT' -or $out.status -eq 'BLOCKED') { break }
        # Multiple identical card-number hits: check the REAL idProduct, not search rank.
        # Limit page visits to avoid excessive requests; unresolved stays for review.
        if ($likely.Count -gt 1 -and $allowedIds.Count -gt 0) {
          $uniqueLinks = @($likely | Group-Object { ([string]$_.url).Split('?')[0] } |
            ForEach-Object { $_.Group[0] })
          $matchesById = @()
          foreach ($link in @($uniqueLinks | Select-Object -First 6)) {
            if ([string]$link.url -notmatch '^https://www[.]cardmarket[.]com/(de|en)/Pokemon/Products/Singles/') { continue }
            if (!(Move-To ([string]$link.url))) { continue }
            $pageProductId = ''
            for ($ii = 0; $ii -lt 12; $ii++) {
              $pageProductId = [string](Js 'document.querySelector("input[name=idProduct]")?.value||""' 10)
              if ($pageProductId -match '^[0-9]+$') { break }
              Start-Sleep -Milliseconds 400
            }
            if ($pageProductId -eq [string]$case.catalog_singleton_id -or
                ([string]$case.routing_state -eq 'VERIFIED_PRODUCT_ID' -and $pageProductId -in $allowedIds)) {
              $matchesById += $link
            }
            if ($matchesById.Count -gt 0) { break }
            Start-Sleep -Seconds ([Math]::Max(3,$DelaySeconds))
          }
          if ($matchesById.Count -eq 1) {
            $likely = @($matchesById[0])
          } else {
            $likely = @($uniqueLinks)
          }
        }
      }

      if ($likely.Count -eq 1) {
        $out.url = $likely[0].url
        $versionMatch=[regex]::Match(([string]$out.url).Split('?')[0],'(?i)(?:^|[-/])V([1-9][0-9]?)(?=[-/]|$)')
        if($versionMatch.Success){$out.observed_url_version='V'+$versionMatch.Groups[1].Value}
        if (!(Move-To $out.url)) { throw 'PRODUCT_PAGE_TIMEOUT' }
        # The last page may have been a Cloudflare interstitial or a login
        # form. Re-check access on the ACTUAL product page, not the previous
        # search page. Never treat an inaccessible account as authenticated.
        $auth = (Js '!!document.querySelector("a[href*=User_Logout],form[action*=User_Logout]")') -and (-not $ForcePublicMode)
        $out.access_mode = if ($auth) {'AUTHENTICATED'} else {'PUBLIC_LIMITED'}
        if ($RequireLogin -and -not $auth) { Assert-CardmarketAuthenticated }
        $candidateId = ''
        for ($pi = 0; $pi -lt 20; $pi++) {
          $candidateId = [string](Js 'document.querySelector("input[name=idProduct]")?.value||""')
          if ($candidateId -match '^[0-9]+$') { break }
          Start-Sleep -Milliseconds 500
        }
        $out.product_id = $candidateId
        if($candidateId -and $guidePrices.ContainsKey($candidateId)){
          $g=$guidePrices[$candidateId]
          $out.official_guide_product_aggregate=@{
            source='OFFICIAL_CARDMARKET_PRODUCT_GUIDE_NOT_VARIANT_PRICE'
            guide_date=$guideDate
            trend=if($g.trend -gt 0){$g.trend}else{$null}
            trend_holo=if($g.'trend-holo' -gt 0){$g.'trend-holo'}else{$null}
            avg30=if($g.avg30 -gt 0){$g.avg30}else{$null}
            avg30_holo=if($g.'avg30-holo' -gt 0){$g.'avg30-holo'}else{$null}
            can_publish_language_print_price=$false
          }
        }
        # Reprints are linked directly from the *opened* product page. Capture
        # the family URL for later independent version-page checks. This adds
        # no Cardmarket navigation and NEVER verifies a foil or price.
        if($candidateId -match '^[0-9]{4,12}$'){
          $familyLink=[string](Js '(()=>{const a=[...document.querySelectorAll("a[href]")].find(x=>{try{const u=new URL(x.href);return u.origin==="https://www.cardmarket.com"&&/^\/(de|en)\/Pokemon\/Cards\/[^/]+\/Versions$/.test(u.pathname)}catch{return false}});return a?a.href.split("?")[0]:""})()')
          if($familyLink -match '^https://www[.]cardmarket[.]com/(de|en)/Pokemon/Cards/[^/]+/Versions$'){
            $out.version_page_url=$familyLink
            $out.version_family_status='PAGE_LINK_FOUND_PRINT_UNVERIFIED'
          }
        }
        $allowed = @([string]$case.cardmarket_ids -split ',' | Where-Object {$_ -match '^[0-9]+$'})
        # The Cardyssey print can lack a product ID while the exact CARD is
        # already known in the independently matched Cardmarket expansion.
        # These are card candidates, NOT verified physical-print identities.
        $cardCandidates=@([string]$case.set_card_candidate_ids -split ',' | Where-Object {$_ -match '^[0-9]+$'})
        $numberEvidence=$false
        $observedPath=([string]$out.url).Split('?')[0]
        # Only the FINAL slug suffix is a candidate collector number.
        # Never accidentally parse -V1 as card #1 in -V1-BS13.
        $numberTail=[regex]::Match($observedPath,'(?i)-[A-Z]{1,8}([0-9]{1,4})$')
        if($numberTail.Success){
          try{$numberEvidence=([int]$numberTail.Groups[1].Value -eq [int]([string]$case.kartennummer))}catch{}
        }
        $out.card_number_url_verified=$numberEvidence
        if (!$candidateId) {
          $out.status = 'NO_PRODUCT_ID'
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif([string]$out.score_matching_status -eq 'SET_NOT_INDEPENDENTLY_VERIFIED'){
          # A perfect-looking number and title still cannot prove the set.
          $out.status='PRODUCT_ID_NOT_VERIFIED_FOR_CARD'
          $out.error='SEARCH_RANK_SET_NOT_INDEPENDENTLY_VERIFIED'
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif($out.score_page_candidate_only -eq $true -and
                 ($out.non_filterable_print_attributes.Count -gt 0 -or [string]$case.catalog_singleton_id -notmatch '^[0-9]{4,12}$')){
          # A scored title is not independent evidence of the exact print.
          # The card/number/actual page are recorded for review only.
          if($numberEvidence -and ([string]$candidateId -in $allowed -or
                                  [string]$candidateId -in $cardCandidates)){
            $out.status='PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED'
            $out.error='SCORED_PAGE_IN_CATALOG_PRINT_NOT_CONFIRMED'
          }else{
            $out.status='PRODUCT_ID_NOT_VERIFIED_FOR_CARD'
            $out.error=if($numberEvidence){'SCORED_PAGE_NOT_IN_CATALOG_CANDIDATES'}else{'SCORED_PAGE_NUMBER_FAILED'}
          }
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif($out.version_family_candidate_used -eq $true){
          # V1/V2 product ID proves the actual page within the observed family.
          # It does not distinguish Pokeball/Masterball or other print details.
          $sameProduct=([string]$candidateId -eq [string]$out.version_family_candidate_id -and
            ([string]$out.url).Split('?')[0] -eq [string]$out.version_family_candidate_url)
          if($sameProduct -and $numberEvidence){
            $out.status='PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED'
            $out.error='VERSION_FAMILY_PAGE_AND_NUMBER_PROVED_PRINT_UNVERIFIED'
          } else {
            $out.status='PRODUCT_ID_NOT_VERIFIED_FOR_CARD'
            $out.error='VERSION_FAMILY_LIVE_PRODUCT_PAGE_MISMATCH'
          }
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif ($setSlugGuards.ContainsKey([string]$case.set_id) -and
                  ([string]$out.url -notmatch ('/Products/Singles/'+[regex]::Escape($setSlugGuards[[string]$case.set_id])+'(?:/|-)'))) {
          $out.status='PRODUCT_SET_SLUG_MISMATCH'
          $out.error='PRODUCT_IS_IN_DIFFERENT_CARDMARKET_EXPANSION'
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif ($out.catalog_direct_id_verified -eq $true -and
                  (([string]$case.foil) -or
                   ([string]$case.stempel -notin @('','[]') -and
                    [string]$case.stempel -notmatch '(?i)1[.] auflage|1st.edition|first.edition') -or
                   ([string]$case.sonderdruck -notin @('','unbegrenzt','unlimited')))) {
          # Exact special art / stamped / rare foil print cannot be proved
          # solely by a page's canonical idProduct. Never reuse the ordinary
          # offer filter price for a hidden physical variant.
          $out.status='PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED'
          $out.error='CATALOG_ID_AND_CARD_NUMBER_VERIFIED_EXACT_PRINT_UNVERIFIED'
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif ($out.reference_page_used -eq $true -and [string]$case.reference_print_verified -eq 'NO' -and [string]$case.reference_standard_filter_supported -ne 'YES') {
          # Direct URL proves the page/print family, not the special edition.
          # A single WP4 product is shared by different movie-stamp variants;
          # Aerodactyl V3 includes more than one prerelease foil pattern.
          # No exact-print median/route may be published from this status.
          if($numberEvidence -and $setSlugGuards.ContainsKey([string]$case.set_id)){
            $out.status='PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED'
            $out.error='EXACT_PRINT_STAMP_OR_FOIL_NEEDS_INDEPENDENT_EVIDENCE'
          }else{
            $out.status='PRODUCT_ID_NOT_VERIFIED_FOR_CARD'
            $out.error='REFERENCE_URL_CARD_NUMBER_OR_SET_UNVERIFIED'
          }
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } elseif ($candidateId -ne [string]$case.catalog_singleton_id -and
                  -not ([string]$case.routing_state -eq 'VERIFIED_PRODUCT_ID' -and $candidateId -in $allowed)) {
          # A product can be proven in the CORRECT CARD/SET but not be proven
          # to match this particular physical edition (1999-2000, 1st Ed,
          # shadowless, reverse, V1/V2). These are different states.
          if(($candidateId -in $cardCandidates -or $candidateId -in $allowed) -and $numberEvidence -and
             $setSlugGuards.ContainsKey([string]$case.set_id)) {
            $out.status = 'PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED'
            $out.error = 'CARD_AND_COLLECTOR_NUMBER_MATCH_PRINT_NEEDS_EVIDENCE'
          } else {
            $out.status = 'PRODUCT_ID_NOT_VERIFIED_FOR_CARD'
            $out.error = 'SET_CARD_NUMBER_OR_PRINT_UNVERIFIED'
          }
          Append-Observation $unverifyPath ($out | ConvertTo-Json -Depth 6 -Compress)
        } else {
          # Identitaet bestaetigt: ID + URL dauerhaft, unabhaengig vom Preis.
          $out.identity_confirmed = $true
          if($out.catalog_direct_id_verified -eq $true -or $out.reference_page_used -eq $true){$out.exact_print_verified=$false}
          if($out.reference_page_used -eq $true -and [string]$case.reference_standard_filter_supported -eq 'YES'){
            $out.standard_offer_filter_candidate=$true
          }
          if (-not $priceNeeded) {
            $out.status = 'MAPPED_NO_PRICE_REQUIRED'
          } else {
            $out.status = 'PRODUCT_PAGE_MATCH_REVIEW'
            $reverseCode = if ([string]$case.drucktyp -match '(?i)^reverse$') { 'Y' } elseif ([string]$case.drucktyp -match '(?i)^(normal|nonholo|common|holo)$') { 'N' } else { '0' }
            $firstCode = if ([string]$case.stempel -match '(?i)1\. auflage|1st.edition|first.edition') { 'Y' } elseif ([string]$case.sonderdruck -match '(?i)^(unbegrenzt|unlimited|copyright 1999-2000|1999-2000-copyright)$') { 'N' } else { '0' }
            $out.expected_reverse_holo = $reverseCode
            $out.expected_first_edition = $firstCode
            # A provider filter cannot distinguish ©1999–2000, special
            # cosmo foil patterns or collection stamps; never claim the
            # filtered result proves that physical special print.
            $out.exact_print_filter_supported=(@($out.non_filterable_print_attributes).Count -eq 0 -and $reverseCode -ne '0' -and $firstCode -ne '0')
            $filterName = 'input[name=\"language[' + $LangCfg.LanguageId + ']\"]'
            $langNames = ($LangCfg.LanguageNames | ForEach-Object { '"'+$_+'"' }) -join ','
            $filterOk=$false
            # Prefer canonical server-side parameters; verify the returned
            # language, condition and print flags before trusting any rows.
            # If the provider does not honor these parameters, restore the
            # product page and use the existing validated form as fallback.
            if($reverseCode -ne '0' -and $directUrlFilterSupported){
              try {
                $targetFilter=New-CardmarketOfferFilterUrl -ProductUrl ([string]$out.url) -LanguageId ([int]$LangCfg.LanguageId) -ReverseCode $reverseCode -FirstEditionCode $firstCode
                $out.direct_filter_attempted=$true

#!/usr/bin/env python3
"""Parse Cardmarket's *observed* /Cards/.../Versions DOM into safe candidate evidence.

The real browser probe runs in a separate tab of the authenticated browser.
A version page can prove a reprint family and its set/number links, NOT the
physical foil, stamp, edition, sale price, or even product ID without opening
the corresponding product page. No automatic price publication or DB writes.
"""
from __future__ import annotations
import argparse
import json
import re
from pathlib import Path
from urllib.parse import urlsplit

BASE = "https://www.cardmarket.com"
SINGLES = re.compile(r"^/(de|en)/Pokemon/Products/Singles/([^/]+)/([^/]+)$")
VERSIONS = re.compile(r"^/(de|en)/Pokemon/Cards/[^/]+/Versions$")
PRODUCT_ID = re.compile(r"^[0-9]{4,12}$")


def safe_url(value: str):
    try:
        u = urlsplit(value)
    except (TypeError, ValueError):
        return None
    if u.scheme != "https" or u.hostname != "www.cardmarket.com" or u.port not in (None, 443):
        return None
    if u.username or u.password:
        return None
    return u


def resolve_version_family(raw: dict, *, card_id: str, set_slug: str,
                           collector_suffix: str, observations: list[dict] = (),
                           catalog: dict | None = None,
                           additional_print_groups: list[dict] = ()) -> dict:
    page = safe_url(raw.get("href", ""))
    if not page or not VERSIONS.fullmatch(page.path):
        raise ValueError("A verified Cardmarket Cards/.../Versions page is required")
    if raw.get("challenge") or raw.get("ready") != "complete":
        raise ValueError("Browser challenge/incomplete page: refuse candidate extraction")
    if not re.fullmatch(r"[A-Za-z0-9-]{2,90}", set_slug):
        raise ValueError("Unsafe set slug")
    if not re.fullmatch(r"[A-Za-z0-9-]{2,32}", collector_suffix):
        raise ValueError("Unsafe collector code")
    groups = [(set_slug, collector_suffix)]
    for group in additional_print_groups:
        gs, gn = str(group.get('setSlug', '')), str(group.get('collectorSuffix', ''))
        if not re.fullmatch(r"[A-Za-z0-9-]{2,90}", gs) or not re.fullmatch(r"[A-Za-z0-9-]{2,32}", gn):
            raise ValueError("Unsafe additional group")
        groups.append((gs, gn))
    expected_path = "/" + page.path.strip("/").split("/")[0] + "/Pokemon/Products/Singles/"
    seen = set()
    all_versions, matched = [], []
    for item in raw.get("anchors", []):
        link = safe_url(item.get("url", ""))
        if not link:
            continue
        m = SINGLES.fullmatch(link.path)
        if not m or (expected_path + m.group(2) + "/" + m.group(3)) != link.path:
            continue
        if link.path in seen:
            continue
        seen.add(link.path)
        product_slug = m.group(3)
        version = re.search(r"(?:^|-)V([1-9][0-9]?)(?:-|$)", product_slug, re.I)
        record = {
            "productUrl": BASE + link.path,
            "setSlug": m.group(2),
            "productSlug": product_slug,
            "version": "V" + version.group(1) if version else None,
            "versionLabel": str(item.get("text", ""))[:160],
            "productId": None,
            "productIdChecked": False,
            "exactPrintVerified": False,
            "priceVerified": False,
        }
        all_versions.append(record)
        if any(m.group(2) == gs and product_slug.upper().endswith("-" + gn.upper()) for gs, gn in groups):
            matched.append(record)
    if not all_versions:
        raise ValueError("No Cardmarket version links visible")
    # A live product-page observation (matching path AND official catalog metadata)
    # is required before recording the exact idProduct. Never guess by V1/V2.
    observations_by_path = {}
    for o in observations:
        u = safe_url(o.get("url", ""))
        pid = str(o.get("id", ""))
        if not u or not SINGLES.fullmatch(u.path) or not PRODUCT_ID.fullmatch(pid):
            continue
        source = next((im.get("src") for im in o.get("imgs", [])
                       if str(im.get("src", "")).startswith("https://product-images.s3.cardmarket.com/")
                       and str(im.get("src", "")).endswith("/" + pid + "/" + pid + ".jpg")), None)
        observations_by_path[u.path] = (pid, source)
    for entry in matched:
        path = urlsplit(entry["productUrl"]).path
        verified = observations_by_path.get(path)
        if verified:
            pid, source_image = verified
            metadata = (catalog or {}).get(pid)
            if catalog is not None and (not metadata or not metadata.get("idExpansion")):
                continue
            entry["productId"] = pid
            entry["productIdChecked"] = True
            if source_image:
                entry["imageUrl"] = source_image
            if metadata:
                entry["expansionId"] = str(metadata["idExpansion"])
                entry["metacardId"] = str(metadata.get("idMetacard", ""))
    # A version family is a candidate set only. Different sets using the same
    # artwork/attacks must not be assigned to this card.
    return {
        "cardId": card_id,
        "versionPageUrl": BASE + page.path,
        "pageHeading": str(raw.get("heading", ""))[:180],
        "totalVersionsObserved": len(all_versions),
        "otherSetVersionsExcluded": len(all_versions) - len(matched),
        "sameSetCardNumberCandidates": matched,
        "candidateCount": len(matched),
        "allCandidatesHaveCheckedProductId": bool(matched) and all(x["productIdChecked"] for x in matched),
        "exactPrintVerified": False,
        "safeToAutoPrice": False,
        "status": "VERSION_FAMILY_PRINT_UNVERIFIED" if matched else "NO_SAME_SET_NUMBER_CANDIDATE",
    }


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--case-file", required=True, help="JSON case with versionsFile, productsFiles, setSlug, collectorSuffix, cardId")
    p.add_argument("--output", required=True)
    a = p.parse_args()
    case = json.loads(Path(a.case_file).read_text(encoding="utf-8-sig"))
    raw = json.loads(Path(case["versionsFile"]).read_text(encoding="utf-8-sig"))
    observations = [json.loads(Path(name).read_text(encoding="utf-8-sig")) for name in case.get("productsFiles", [])]
    catalog = None
    if case.get("catalogFile"):
        feed = json.loads(Path(case["catalogFile"]).read_text(encoding="utf-8-sig"))
        catalog = {str(x["idProduct"]): x for x in feed["products"]}
    result = resolve_version_family(raw, card_id=case["cardId"], set_slug=case["setSlug"],
        collector_suffix=case["collectorSuffix"], observations=observations, catalog=catalog,
        additional_print_groups=case.get("additionalPrintGroups", []))
    Path(a.output).parent.mkdir(parents=True, exist_ok=True)
    Path(a.output).write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    print(json.dumps({"cardId": result["cardId"], "familyVersions": result["totalVersionsObserved"],
        "matched": result["candidateCount"], "allIdChecked": result["allCandidatesHaveCheckedProductId"],
        "status": result["status"]}, ensure_ascii=False))


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Conservative Cardmarket print resolution and language-specific offer pricing.

Input must come from an independently verified Cardmarket product-page / catalog
record. An image hash, V1/V2, a high price, or an unlabelled JP reverse does
NOT independently identify a print. This module NEVER writes price caches.
No external dependencies; optional Pillow/imagehash support is diagnostic.
"""
from __future__ import annotations
import argparse
import dataclasses
from decimal import Decimal, InvalidOperation, ROUND_HALF_UP
from collections import Counter
from dataclasses import dataclass, field
import io
import json
import re
import statistics
from pathlib import Path
from urllib.parse import urlsplit

PRINTS={"normal","reverse","pokeball","masterball","stamp","holo","cosmos","unknown"}
LANGUAGES={"de","en","ja","ko","zh-cn","zh-tw"}
CONDITIONS={"NM","MT","NEAR MINT","MINT"}
BALL_PATTERNS={
 "masterball":re.compile(r"(?i)(?<!\w)(?:master\s*ball|meister\s*ball|masterball|meisterball)(?!\w)"),
 "pokeball":re.compile(r"(?i)(?<!\w)(?:pok[eé]\s*ball|pok[eé]ball)(?!\w)"),
}
NEGATIONS=re.compile(r"(?i)(?:\b(?:no|not|kein(?:e|en|er|es)?|ohne|non|nicht|sin|sans)\b|\b(?:wrong|falsch)\b)")
NEGATED_BALL=re.compile(r"(?i)\b(?:no|not|kein(?:e|en|er|es)?|ohne|non|nicht|sin|sans)\s+(?:(?:a|ein|eine|an)\s+)?(?:master\s*ball|meister\s*ball|masterball|pok[eé]\s*ball|pok[eé]ball)")
STAMP_WORDS={
 "prerelease":re.compile(r"(?i)\b(?:prerelease|pre.release|vorabversion)\b"),
 "play-pokemon":re.compile(r"(?i)\b(?:play!?\s*pok[eé]mon|play!\s*pokemon)\b"),
 "professor-program":re.compile(r"(?i)\b(?:professor.?s?\s+program(?:me)?|professorenprogramm)\b"),
 "cosmos":re.compile(r"(?i)\bcosmos\s*(?:holo|foil)?\b"),
}
def labels(text:str)->set[str]:
    text=str(text or "")
    out={key for key,rx in BALL_PATTERNS.items() if rx.search(text)}
    out|={key for key,rx in STAMP_WORDS.items() if rx.search(text)}
    return out

def classified_comment(comment:str, target:str)->str:
    """'positive', 'contradiction', 'missing', 'ambiguous'.

    An absent comment never proves a special print, even in JP 151.
    Bare PB/MB abbreviations are deliberately NOT accepted as evidence.
    """
    target=str(target or "").casefold()
    found=labels(comment)
    if not comment or not comment.strip():return "missing"
    if NEGATED_BALL.search(comment):return "ambiguous"
    if target in {"pokeball","masterball"}:
        other={"pokeball","masterball"}-{target}
        if found&other:return "contradiction"
        if target in found:return "positive"
        return "missing"
    if target.startswith("stamp:"):
        stamp=target.partition(":")[2]
        if stamp in found and not NEGATIONS.search(comment):return "positive"
        return "missing"
    return "not_applicable"

@dataclass(frozen=True)
class Product:
    product_id:str
    set_slug:str
    product_url:str
    number:str
    name:str=""
    title:str=""
    version:str=""
    price_trend:float|None=None
    image_url:str|None=None
    confirmed_product_page:bool=False
    official_expansion_id:str|None=None

@dataclass
class Decision:
    product_id:str
    print_type:str="unknown"
    confidence:str="LOW"
    evidence:list[str]=field(default_factory=list)
    review_required:bool=True
    safe_to_publish_price:bool=False
    # HIGH refers to identity only; price requires independent offer validation

def valid_product_url(u:str)->bool:
    try:x=urlsplit(u)
    except (TypeError,ValueError):return False
    return bool(x.scheme=="https" and x.hostname=="www.cardmarket.com" and
      x.port in (None,443) and not x.username and not x.password and
      re.fullmatch(r"/(?:de|en)/Pokemon/Products/Singles/[A-Za-z0-9-]+/[A-Za-z0-9-]+",x.path))

def classify_product(product:Product,*,valid_set_slugs:set[str],
                     matched_number:str, confirmed_print_evidence:str|None=None,
                     foil_image_evidence:dict|None=None)->Decision:
    """Classify a page only if set+number were independently validated.

    A high-cost product is not evidence of a Master Ball. A generic phash
    cannot distinguish shared artwork with different foil; image proof must
    explicitly be 'human_verified' and identify the printed feature.
    """
    result=Decision(product_id=str(product.product_id))
    if not product.confirmed_product_page or not valid_product_url(product.product_url):
        result.evidence.append("PRODUCT_PAGE_NOT_VERIFIED");return result
    if product.set_slug not in valid_set_slugs or not valid_set_slugs:
        result.evidence.append("SET_NOT_VERIFIED");return result
    a=re.search(r"0*(\d+)$",str(product.number))
    b=re.search(r"0*(\d+)$",str(matched_number))
    if not a or not b or a.group(1)!=b.group(1):
        result.evidence.append("NUMBER_MISMATCH");return result
    result.evidence+=["SET_VERIFIED","COLLECTOR_NUMBER_VERIFIED","PRODUCT_PAGE_VERIFIED"]
    # A Trainer card can itself be named "Master Ball". Only a structured
    # provider version/print field, not the card name/title, is foil evidence.
    explicit=labels(product.version)
    if "masterball" in explicit and "pokeball" in explicit:
        result.evidence.append("CONFLICTING_PRINT_LABELS");return result
    if "masterball" in explicit or "pokeball" in explicit:
        result.print_type="masterball" if "masterball" in explicit else "pokeball"
        result.confidence="HIGH"
        result.evidence.append("EXPLICIT_PROVIDER_PRINT_LABEL")
        result.review_required=False
        return result
    if confirmed_print_evidence in PRINTS-{"unknown"}:
        # Human/manual evidence must be supplied by a verified review action,
        # not by untrusted comments or price heuristics.
        result.print_type=confirmed_print_evidence
        result.confidence="HIGH";result.review_required=False
        result.evidence.append("HUMAN_CONFIRMED_PRINT")
        return result
    if foil_image_evidence:
        if (foil_image_evidence.get("method")=="human_verified_roi"
                and foil_image_evidence.get("subjectProductId")==str(product.product_id)
                and foil_image_evidence.get("print") in PRINTS-{"unknown"}
                and foil_image_evidence.get("reviewer")):
            result.print_type=foil_image_evidence["print"]
            result.confidence="HIGH";result.review_required=False
            result.evidence.append("HUMAN_VERIFIED_FOIL_REGION")
            return result
        result.evidence.append("IMAGE_SIMILARITY_ONLY_NOT_FOIL_PROOF")
    result.evidence.append("V1_V2_OR_PRICE_NOT_PRINT_PROOF")
    return result

def phash_distance(a:bytes,b:bytes)->int|None:
    """Optional diagnostic; near-equal artwork != equal foil."""
    try:
        from PIL import Image
        import imagehash
        x=imagehash.phash(Image.open(io.BytesIO(a)).convert("RGB"))
        y=imagehash.phash(Image.open(io.BytesIO(b)).convert("RGB"))
        return int(x-y)
    except (ImportError,OSError,ValueError,TypeError):
        return None

def price_hint(products:list[Product])->dict:
    """Trend-price only ranks candidates for manual review; never maps prints."""
    valid=sorted([p for p in products if p.price_trend and p.price_trend>0],
                 key=lambda p:p.price_trend or 0)
    ratio=(valid[-1].price_trend/valid[0].price_trend) if len(valid)==2 else None
    return {"orderedProductIds":[p.product_id for p in valid],
            "priceRatio":round(ratio,2) if ratio else None,
            "reviewSuggestion":"CHEAPER_POKEBALL_MORE_EXPENSIVE_MASTERBALL_POSSIBLE" if ratio and ratio>=2.5 else None,
            "printVerified":False,"safeToAutoPublish":False}

def parse_money(value)->Decimal:
    if isinstance(value,bool):raise ValueError("price")
    if isinstance(value,str):
        value=value.replace("€","").replace("\u00a0","").replace(" ","").strip()
        if not re.fullmatch(r"[0-9.,]+",value):raise ValueError("price")
        if "," in value and "." in value:
            value=value.replace(".","").replace(",",".") if value.rfind(",")>value.rfind(".") else value.replace(",","")
        elif "," in value:value=value.replace(",",".")
    try:amount=Decimal(str(value))
    except (InvalidOperation,TypeError,ValueError):raise ValueError("price")
    if not amount.is_finite() or amount<=0:raise ValueError("price")
    return amount

def percentile(values:list[Decimal],p:Decimal)->Decimal:
    values=sorted(values)
    i=Decimal(len(values)-1)*p
    low=int(i)
    return values[low]+(values[min(low+1,len(values)-1)]-values[low])*(i-Decimal(low))

def price_diagnostic(data:dict)->dict:
    """Read-only robust asking median; requires complete list for certification.

    Same idProduct, language, condition, special print positive comments where
    required, reverse/edition flags, no duplicates. IQR only with >=8 offers.
    No automatic cross-language conversion or price-feed publication.
    """
    exp=data.get("expected") or {}
    target=str(exp.get("print") or "").casefold()
    lang=exp.get("language")
    pid=str(exp.get("product_id") or "")
    if lang not in LANGUAGES or target not in PRINTS-{"unknown"} or not pid.isdigit():
        raise ValueError("language, print and numeric product_id required")
    rows=data.get("offers")
    if not isinstance(rows,list):raise ValueError("offers must be list")
    accepted=[];excluded=Counter();seen=set()
    for row in rows:
        if not isinstance(row,dict):
            excluded["INVALID_ROW"]+=1;continue
        oid=str(row.get("offer_id") or "")
        if not oid or oid in seen:
            excluded["DUPLICATE_OR_MISSING_OFFER_ID"]+=1;continue
        seen.add(oid)
        if str(row.get("product_id"))!=pid:
            excluded["WRONG_PRODUCT"]+=1;continue
        if row.get("language")!=lang:
            excluded["WRONG_LANGUAGE"]+=1;continue
        if str(row.get("condition") or "").upper() not in CONDITIONS:
            excluded["NOT_NM_OR_BETTER"]+=1;continue
        if row.get("signed") is not False or row.get("altered") is not False or row.get("graded") is not False:
            excluded["MODIFIED_OR_UNVERIFIED"]+=1;continue
        if row.get("bundle") is not False:
            excluded["BUNDLE_OR_UNVERIFIED"]+=1;continue
        reverse=row.get("is_reverse")
        if not isinstance(reverse,bool):
            excluded["UNKNOWN_REVERSE_FLAG"]+=1;continue
        if target=="normal" and reverse or target=="reverse" and not reverse:
            excluded["WRONG_REVERSE_FLAG"]+=1;continue
        if target in {"pokeball","masterball"}:
            comment_status=classified_comment(row.get("comment") or "",target)
            if comment_status!="positive":
                excluded["PRINT_COMMENT_"+comment_status.upper()]+=1;continue
            if lang=="ja" and not reverse:
                excluded["JAPANESE_SPECIAL_REQUIRES_REVERSE"]+=1;continue
        if target=="stamp" and not (exp.get("confirmed_print_identity") is True):
            excluded["STAMP_TYPE_UNVERIFIED"]+=1;continue
        try:accepted.append(parse_money(row["price_eur"]))
        except (KeyError,ValueError):excluded["INVALID_PRICE"]+=1
    complete_info=data.get("pagination") or {}
    all_complete=(complete_info.get("all_pages_fetched") is True
                  and complete_info.get("capped") is False
                  and isinstance(complete_info.get("total_offers"),int)
                  and complete_info["total_offers"]==len(seen)
                  and excluded["DUPLICATE_OR_MISSING_OFFER_ID"]==0)
    med=statistics.median(accepted) if accepted else None
    cleaned=list(accepted)
    outlier_count=0
    if len(cleaned)>=8:
        q1=percentile(cleaned,Decimal("0.25"))
        q3=percentile(cleaned,Decimal("0.75"))
        iqr=q3-q1
        if iqr>0:
            low=q1-Decimal("1.5")*iqr
            high=q3+Decimal("1.5")*iqr
            cleaned=[x for x in cleaned if low<=x<=high]
            outlier_count=len(accepted)-len(cleaned)
    cleaned_median=statistics.median(cleaned) if cleaned else None
    trend=data.get("product_wide_trend")
    ratio=None
    try:
        tr=parse_money(trend)
        if cleaned_median is not None:ratio=float(cleaned_median/tr)
    except ValueError:pass
    result={
       "basis":"ACTIVE_ASKING_NOT_SOLD",
       "language":lang,"product_id":pid,"print":target,
       "total_rows":len(rows),"unique_offers":len(seen),
       "accepted_offers":len(accepted),"rejected":dict(excluded),
       "all_pages_complete":all_complete,"iqr_removed":outlier_count,
       "median_all_accepted_eur":str(med.quantize(Decimal(".01"),rounding=ROUND_HALF_UP)) if med else None,
       "median_iqr_diagnostic_eur":str(cleaned_median.quantize(Decimal(".01"),rounding=ROUND_HALF_UP)) if cleaned_median else None,
       "product_guide_ratio":round(ratio,2) if ratio is not None else None,
       "guide_outlier_warning":ratio is not None and (ratio<0.5 or ratio>2),
       "exact_print_verified":exp.get("confirmed_print_identity") is True,
       "safe_to_publish":bool(all_complete and accepted and exp.get("confirmed_print_identity") is True
          and not excluded["PRINT_COMMENT_AMBIGUOUS"]
          and (target not in {"pokeball","masterball"} or all_complete)),
    }
    return result

def main():
    a=argparse.ArgumentParser()
    a.add_argument("input",type=Path)
    a.add_argument("--output",type=Path)
    args=a.parse_args()
    data=json.loads(args.input.read_text(encoding="utf-8-sig"))
    if data.get("operation")=="price":result=price_diagnostic(data)
    else:
        p=Product(**data["product"])
        d=classify_product(p,valid_set_slugs=set(data.get("verified_set_slugs",[])),
            matched_number=data["card_number"],confirmed_print_evidence=data.get("manual_print"),
            foil_image_evidence=data.get("foil_image_evidence"))
        result=dataclasses.asdict(d)
    rendered=json.dumps(result,ensure_ascii=False,indent=2)+"\n"
    if args.output:args.output.write_text(rendered,encoding="utf-8")
    print(rendered,end="")
if __name__=="__main__":main()

# Conservative candidate ranking. A score cannot confirm a physical print,
# language-specific price, or product ID. Caller must verify the actual page.

function Resolve-CardmarketSearchCandidates {
 param(
   [object[]]$Links,
   [string]$PrintedNumber,
   [string]$SetSlug = '',
   [string]$Name = '',
   [string]$Stamp = '',
   [string]$Foil = '',
   [string]$PrintType = '',
   [string[]]$AllowedProductIds = @()
 )
 $ranked=[Collections.Generic.List[object]]::new()
 $seen=[Collections.Generic.HashSet[string]]::new([StringComparer]::OrdinalIgnoreCase)
 $expected=$PrintedNumber.Trim()
 foreach($link in @($Links)){
   $uri=$null
   try{$uri=[uri]([string]$link.url)}catch{continue}
   if(!$uri -or $uri.Scheme -ne 'https' -or $uri.Host -ne 'www.cardmarket.com') {continue}
   if($uri.AbsolutePath -notmatch '^/(de|en)/Pokemon/Products/Singles/([A-Za-z0-9-]+)/([A-Za-z0-9-]+)$'){continue}
   $actualSet=[string]$Matches[2];$productSlug=[string]$Matches[3]
   $path=$uri.GetLeftPart([UriPartial]::Path)
   if(!$seen.Add($path)){continue}
   # A verified expansion guard is a hard requirement, not a score bonus.
   if($SetSlug -and $actualSet -cne $SetSlug){continue}
   # The final collector token is the ONLY part allowed to satisfy the
   # printed number. A title like '113' never matches collector #13.
   $tail=[regex]::Match($productSlug,'(?i)(?:^|-)([A-Za-z]{0,12})([0-9]{1,4})$')
   if(!$tail.Success){continue}
   $num=[string]$tail.Groups[2].Value
   if($expected -match '^\d{1,4}$'){
     if([int]$expected -ne [int]$num){continue}
   } elseif($expected -match '^[A-Za-z]{1,12}0*[0-9]{1,4}$'){
     # Promo-style printed identifiers e.g. DP05 or SVP099.
     $expectedCode=$expected.ToUpperInvariant().TrimStart()
     if(($tail.Groups[1].Value+$num).ToUpperInvariant() -cne $expectedCode){continue}
   } else {continue}
   $title=[string]$link.text
   $searchText=($title+' '+$productSlug).ToLowerInvariant()
   $score=100
   if($SetSlug){$score+=40}
   if($Stamp.Trim()){
     $stampText=$Stamp -replace '^[\[\]"\s]+|[\[\]"\s]+$',''
     if($stampText -and $searchText.Contains($stampText.ToLowerInvariant())){$score+=12}
     # 'Promo' alone does NOT prove a specific stamp.
   } elseif($actualSet -match '(?i)promo' -and !$SetSlug){$score-=3}
   if($Foil.Trim() -and $searchText.Contains($Foil.ToLowerInvariant())){$score+=5}
   if($PrintType -match '(?i)^reverse$'){
     if($searchText.Contains('reverse holo')){$score+=3}
   } elseif($searchText.Contains('reverse holo')){$score-=3}
   if($Name.Trim() -and $title.ToLowerInvariant().Contains($Name.ToLowerInvariant())){$score+=2}
   # V2 and bracketed attack names are NOT evidence of a stamp.
   $ranked.Add([pscustomobject]@{url=$path;text=$title;setSlug=$actualSet;
      printedNumber=$num;score=$score;productSlug=$productSlug})
 }
 $ordered=@($ranked | Sort-Object @{Expression='score';Descending=$true}, @{Expression='url';Descending=$false})
 $winner=$null
 $reason='NO_NUMBER_MATCH'
 if($ordered.Count -gt 0) {
   if(!$SetSlug){$reason='SET_NOT_INDEPENDENTLY_VERIFIED'}
   elseif($ordered.Count -eq 1){$winner=$ordered[0];$reason='UNIQUE_SET_NUMBER_PAGE_CANDIDATE'}
   elseif([int]$ordered[0].score -ge [int]$ordered[1].score+10){
     # Even a large score gap is only a candidate to inspect, NOT print proof.
     $winner=$ordered[0];$reason='HIGHER_SCORING_SET_NUMBER_PAGE_CANDIDATE'
   }else{$reason='MULTIPLE_SAME_SET_NUMBER_CANDIDATES'}
 }
 return [pscustomobject]@{
    candidates=$ordered;uniquePageCandidate=$winner;reason=$reason;
    candidateCount=$ordered.Count;exactPrintVerified=$false;priceVerified=$false
 }
}

# Canonical, first-party Cardmarket offer-filter URLs. Never assume that
# query parameters alone prove filtering; the caller must validate the DOM.
function New-CardmarketOfferFilterUrl {
 param([string]$ProductUrl,[ValidateSet(1,3)][int]$LanguageId,
       [ValidateSet('Y','N','0')][string]$ReverseCode='0',
       [ValidateSet('Y','N','0')][string]$FirstEditionCode='0')
 $u=$null
 try{$u=[uri]$ProductUrl}catch{throw 'INVALID_FILTER_BASE_URL'}
 if(!$u -or $u.Scheme -ne 'https' -or $u.Host -ne 'www.cardmarket.com' -or
    $u.AbsolutePath -notmatch '^/(de|en)/Pokemon/Products/Singles/[A-Za-z0-9-]+/[A-Za-z0-9-]+$'){
    throw 'INVALID_FILTER_BASE_URL'
 }
 $country=if($LanguageId -eq 3){'de'}else{'en'}
 if($u.AbsolutePath -notmatch ('^/'+$country+'/Pokemon/')){throw 'FILTER_LANGUAGE_PATH_MISMATCH'}
 $q='language='+$LanguageId+'&minCondition=2&isSigned=N&isAltered=N'
 if($ReverseCode -ne '0'){$q+='&isReverseHolo='+$ReverseCode}
 if($FirstEditionCode -ne '0'){$q+='&isFirstEd='+$FirstEditionCode}
 return $u.GetLeftPart([UriPartial]::Path)+'?'+$q
}

# Cardmarket seller remarks are diagnostic evidence, not an automatic print ID.
# In particular an absent comment does not mean Pokeball for Japanese reverse.
function Get-CardmarketOfferCommentClassification {
 param([string]$Comment,[string]$TargetPrint)
 $text=[string]$Comment
 $target=[string]$TargetPrint
 if([string]::IsNullOrWhiteSpace($text)){return 'MISSING'}
 $mb=($text -match '(?i)(?<!\w)(?:master\s*ball|meister\s*ball|masterball|meisterball)(?!\w)')
 $pb=($text -match '(?i)(?<!\w)(?:pok[eé]\s*ball|pok[eé]ball)(?!\w)')
 $no=($text -match '(?i)\b(?:no|not|kein(?:e|en|er|es)?|ohne|non|nicht|sin|sans)\s+(?:(?:a|ein|eine|an)\s+)?(?:master\s*ball|meister\s*ball|masterball|pok[eé]\s*ball|pok[eé]ball)')
 if($no){return 'AMBIGUOUS'}
 if($target -eq 'masterball'){
   if($pb){return 'CONTRADICTION'}
   if($mb){return 'POSITIVE'}
   return 'MISSING'
 }
 if($target -eq 'pokeball'){
   if($mb){return 'CONTRADICTION'}
   if($pb){return 'POSITIVE'}
   return 'MISSING'
 }
 if($target -like 'stamp:*'){
   $stamp=$target.Substring(6)
   switch($stamp){
     'prerelease' {$ok=($text -match '(?i)\b(?:prerelease|pre.release)\b')}
     'professor-program' {$ok=($text -match '(?i)\b(?:professor.?s?\s+program(?:me)?|professorenprogramm)\b')}
     'play-pokemon' {$ok=($text -match '(?i)\bplay!?\s*pok[eé]mon\b')}
     'cosmos' {$ok=($text -match '(?i)\bcosmos\b')}
     default {$ok=$false}
   }
   if($ok){return 'POSITIVE'}
   return 'MISSING'
 }
 return 'NOT_APPLICABLE'
}

#!/usr/bin/env python3
"""Official catalog metacard-family *candidate* index for the full DE/EN queue.

idMetacard groups printings including separate Additional expansions, but
does not certify Cardyssey identity, collector number, foil or price. Index
keeps original official expansion IDs to avoid cross-set autopromotion.
"""
import argparse,csv,hashlib,json,os,re,tempfile
from collections import defaultdict,Counter
from pathlib import Path
ROOT=Path('/home/tv/cardshop')
BASE=Path('/mnt/c/Users/TV/AppData/Local/Cardyssey/CardmarketCollector')
URL='https://downloads.s3.cardmarket.com/productCatalog/productList/products_singles_6.json'
FEEDS=Path('/home/tv/.local/state/cardyssey-queue/price-feeds')
def build(base=BASE,feeds=FEEDS):
    p=feeds/(hashlib.sha256(URL.encode()).hexdigest()+'.json')
    products=json.loads(p.read_text(encoding='utf-8'))['products']
    by_id={str(x['idProduct']):x for x in products if x.get('idProduct')}
    by_meta=defaultdict(list)
    for x in products:
        if x.get('idMetacard'):by_meta[str(x['idMetacard'])].append(x)
    result={};summary=Counter()
    for lang in ('de','en'):
        cards=defaultdict(lambda:{'setId':None,'seeds':set(),'variants':0})
        with (base/f'queue-{lang}.csv').open(encoding='utf-8-sig',newline='') as f:
            for r in csv.DictReader(f):
                card=r['cardyssey_karte'];d=cards[card];d['setId']=r['set_id'];d['variants']+=1
                if r.get('candidate_status')=='SET_EXPANSION_AMBIGUOUS':continue
                for val in (str(r.get('cardmarket_ids') or '')+','+
                            str(r.get('set_card_candidate_ids') or '')+','+
                            str(r.get('catalog_singleton_id') or '')).split(','):
                    if re.fullmatch('[0-9]{4,12}',val) and val in by_id:d['seeds'].add(val)
        for card,d in cards.items():
            possible=defaultdict(set)
            for pid in d['seeds']:
                meta=str(by_id[pid].get('idMetacard') or '')
                if meta and meta!='0':possible[meta].add(pid)
            if not possible:continue
            # Distinct families for a single card are ambiguous and cannot
            # safely be merged: retain them as separate, unverified options.
            meta_groups=[]
            for meta,seeds in sorted(possible.items()):
                members=by_meta.get(meta,[])
                if len(members)<2:continue
                meta_groups.append({'metacardId':meta,'seedProductIds':sorted(seeds),
                  'candidateProducts':[{'id':str(x['idProduct']),
                     'expansionId':str(x.get('idExpansion') or ''),
                     'name':str(x.get('name') or '')[:170]}
                    for x in members[:30]],'truncated':len(members)>30,
                  'multipleExpansions':len({str(x.get('idExpansion')) for x in members})>1,
                  'printIdentityVerified':False})
            if meta_groups:
                result[lang+'|'+card]={'cardId':card,'language':lang,'setId':d['setId'],
                    'seedIsVerifiedPrint':False,'families':meta_groups}
                summary[lang+'_cards']+=1
                if len(meta_groups)>1:summary[lang+'_multipleFamilies']+=1
                if any(x['multipleExpansions'] for x in meta_groups):summary[lang+'_crossExpansionFamilies']+=1
    return {'source':'CARDMARKET_OFFICIAL_PRODUCT_CATALOG_METACARD_ONLY',
            'printIdentityVerified':False,'pricesPublished':0,
            'summary':dict(summary),'cards':result}
def write(data,path:Path):
    path.parent.mkdir(parents=True,exist_ok=True)
    fd,temp=tempfile.mkstemp(prefix='cm-family-',suffix='.json.tmp',dir=path.parent)
    try:
        with os.fdopen(fd,'w',encoding='utf-8') as f:
            json.dump(data,f,ensure_ascii=False,separators=(',',':'))
        os.replace(temp,path)
    finally:
        if os.path.exists(temp):os.unlink(temp)
if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--output',type=Path,required=True)
    a=p.parse_args();x=build();write(x,a.output)
    print(json.dumps({'summary':x['summary'],'indexedCards':len(x['cards']),
       'printVerified':False,'pricesPublished':0},ensure_ascii=False))

def pending_queries(data):

    for cid,item in data.items():

            pid=str(v.get('productId') or '')

def explain_identity(row,products):
    """Only evidence-supported reasons, not guesses based on catalog ID or date."""
    ids=[str(v) for v in (row.get('matched_set_ids') or [])]
    candidates=[]
    for pid in ids:
        raw=products.get(pid,{})
        candidates.append({'id':pid,'name':str(raw.get('name') or 'Name nicht verfügbar'),
            'expansionId':str(raw.get('idExpansion') or ''),
            'metacardId':str(raw.get('idMetacard') or ''),
            'addedAt':str(raw.get('dateAdded') or '')})
    classification=str(row.get('classification') or '')
    special=any([str(row.get('foil') or '').strip(),str(row.get('subtype') or '').strip(),
                str(row.get('stamps') or '').strip() not in ('','[]','null'),
                'REVERSE' in str(row.get('print_type') or '').upper(),
                'HOLO' in str(row.get('print_type') or '').upper()])
    same_metacard=len(candidates)>1 and len({x['metacardId'] for x in candidates})==1 and bool(candidates[0]['metacardId'])
    if len(ids)>1:
        reason=f"{len(ids)} Cardmarket-Produkte gehören laut Katalog zum gleichen zugeordneten Set, aber die gedruckte Kartennummer ({row.get('printed_number') or '?'}) ist für keine dieser Produkt-IDs durch den offiziellen Feed bestätigt."
        if same_metacard: reason+=' Sie teilen sogar dieselbe Metakarten-ID; der Produktfeed nennt keinen belastbaren Druck-Unterschied.'
        else: reason+=' Der Produktfeed löst die Druckvariante nicht eindeutig auf.'
        action='Jeden Kandidaten über seinen ID-Direktlink öffnen; Set-Kürzel, Kartennummer, V1/V2, Illustration und tatsächliche Angebotssprache vergleichen. Wenn die Produktseite keine Unterscheidung zulässt, offenlassen.'
        kind='CATALOG_DUPLICATE'
    elif len(ids)==1 and special:
        reason='Ein Produkt ist dem Set zugeordnet, aber Holo, Reverse, Edition oder Sonderdruck wurde durch die Quelle nicht bestätigt.'
        action='Prüfen, ob genau die angezeigte Druckvariante auf der Cardmarket-Produktseite belegbar ist. Sonst offenlassen.'
        kind='PRINT_UNVERIFIED'
    elif 'SET_NOT_VERIFIED' in classification or 'INFERRED_SET' in classification:
        reason='Das Cardmarket-Set ist noch nicht unabhängig und eindeutig mit dem Cardyssey-Set abgeglichen.'
        action='Setname und Set-Kartennummer abgleichen. Ohne bestätigtes Set keine Produkt-ID zuweisen.'
        kind='SET_UNVERIFIED'
    elif not ids:
        reason='Es liegt kein bestätigter Produktkandidat im richtigen Cardmarket-Set vor.'
        action='Hier gibt es derzeit keine sichere Auswahl; die automatische Suche muss zuerst neue Kandidaten liefern.'
        kind='MISSING_MATCH'
    else:
        reason='Der genaue Kartendruck wurde noch nicht durch eine geeignete Quelle bestätigt.'
        action='Produktseite mit Set, Nummer, Sprache und Druck abgleichen; nur dann bestätigen.'
        kind='PRINT_UNVERIFIED'
    return {'kind':kind,'reason':reason,'action':action,
       'proofAvailable':False, 'sameMetacard':same_metacard,'candidates':candidates}

def sync_queue_identity(language, card_id, variant_id, product_id, base=Path('/mnt/c/Users/TV/AppData/Local/Cardyssey/CardmarketCollector'), product_url=None):
    """Make confirmed DB identity visible to next CSV-based browser run.

    Atomic replacement keeps ongoing scans reading their original complete CSV.
    Unverified price data is never changed.
    """
    if language not in ALLOWED_LANG:return False
    path=base/f'queue-{language}.csv'
    if not path.is_file():return False
    lock=base/f'queue-{language}.review.lock'
    with lock.open('a') as lk:
        fcntl.flock(lk,fcntl.LOCK_EX)
        with path.open(encoding='utf-8-sig',newline='') as f:
            reader=csv.DictReader(f)
            fields=reader.fieldnames
            rows=list(reader)
        if not fields or 'routing_state' not in fields:return False
        if product_url and 'manual_scan_requested_at' not in fields:fields.append('manual_scan_requested_at')
        updated=0
        for row in rows:
            if row.get('cardyssey_karte')!=card_id or row.get('variante_id')!=variant_id:continue
            if row.get('cardmarket_ids')!=product_id:
                row['known_url']=''  # Never retain a URL for a different printing.
            row['cardmarket_ids']=product_id
            row['routing_state']='VERIFIED_PRODUCT_ID'
            if product_url and product_url.startswith('https://www.cardmarket.com/'):
                row['known_url']=product_url
                row['manual_scan_requested_at']=__import__('datetime').datetime.now(__import__('datetime').timezone.utc).isoformat()
            updated+=1
        if updated!=1:return False
        tmp=path.with_suffix(path.suffix+'.review.tmp')
        try:
            with tmp.open('w',encoding='utf-8-sig',newline='') as f:
                writer=csv.DictWriter(f,fieldnames=fields)
                writer.writeheader()
                writer.writerows(rows)
            os.replace(tmp,path)
        finally:tmp.unlink(missing_ok=True)
        return True
def revert_queue_identity(language, card_id, variant_id, candidate_ids,
                          base=Path('/mnt/c/Users/TV/AppData/Local/Cardyssey/CardmarketCollector')):
    """Remove only manually approved CSV routing, retain catalog suggestions."""

    path=base/f'queue-{language}.csv'

    lock=base/f'queue-{language}.review.lock'

        with path.open(encoding='utf-8-sig',newline='') as f:
            rd=csv.DictReader(f);fields=list(rd.fieldnames or []);rows=list(rd)
        n=0
        for row in rows:

            row['routing_state']='UNVERIFIED'

            row['known_url']=''
            if 'manual_scan_requested_at' in fields:row['manual_scan_requested_at']=''

        tmp=path.with_suffix('.csv.undo.tmp')
        try:

                w=csv.DictWriter(f,fieldnames=fields);w.writeheader();w.writerows(rows)
            os.replace(tmp,path)
        finally:tmp.unlink(missing_ok=True)
    return True

def undo_case(data):
    lang=data.get('language');card=data.get('cardId');variant=data.get('variantId')
    if lang not in ALLOWED_LANG or not all(isinstance(x,str) and re.fullmatch(r'[a-zA-Z0-9_.:-]{1,128}',x) for x in (card,variant)):
        raise ValueError('invalid identity')
    # Snapshot and revoke a human decision atomically. Never delete an
    # unrelated provider or an independently verified price. The catalogue
    # candidate remains available as an UNVERIFIED suggestion.
    sql=f"""BEGIN;
CREATE TABLE IF NOT EXISTS cardmarket_review_undo_log (
 id BIGSERIAL PRIMARY KEY,undone_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
 language_code TEXT NOT NULL,card_id TEXT NOT NULL,variant_id TEXT NOT NULL,
 previous_status TEXT NOT NULL,previous_product_id TEXT,
 previous_note TEXT,deleted_human_route JSONB);
WITH prior AS (
 SELECT language_code,card_id,variant_id,status,selected_product_id,reviewer_note
 FROM cardmarket_print_identity_review WHERE language_code={quote(lang)}
 AND card_id={quote(card)} AND variant_id={quote(variant)}
 AND status IN ('CONFIRMED_PRINT','NO_MATCH','DEFERRED') FOR UPDATE
), removed AS (
 DELETE FROM price_source_routing r USING prior p
 WHERE r.language_code=p.language_code AND r.card_id=p.card_id AND r.variant_id=p.variant_id
 AND p.status='CONFIRMED_PRINT' AND r.primary_provider='CARDMARKET'
 AND r.verification_state='VERIFIED_PRODUCT_ID'
 AND r.source_evidence LIKE 'Human confirmed exact%'
 AND r.provider_product_id=p.selected_product_id
 RETURNING to_jsonb(r) AS previous_route
), log AS (
 INSERT INTO cardmarket_review_undo_log
 (language_code,card_id,variant_id,previous_status,previous_product_id,previous_note,deleted_human_route)
 SELECT p.language_code,p.card_id,p.variant_id,p.status,p.selected_product_id,p.reviewer_note,
 (SELECT previous_route FROM removed LIMIT 1) FROM prior p RETURNING id
), restored AS (
 UPDATE cardmarket_print_identity_review r SET status='PENDING_REVIEW',
 selected_product_id=NULL,reviewer_note=NULL,reviewed_at=NULL
 FROM prior p WHERE r.language_code=p.language_code AND r.card_id=p.card_id
 AND r.variant_id=p.variant_id AND EXISTS(SELECT 1 FROM log)
 RETURNING r.matched_set_ids
)
SELECT json_build_object('undone',(SELECT COUNT(*) FROM restored),
 'routeRemoved',(SELECT COUNT(*) FROM removed),
 'candidateIds',(SELECT matched_set_ids FROM restored LIMIT 1))::text;
COMMIT;"""
    output=pg(sql)
    records=[json.loads(line) for line in output.splitlines() if line.startswith('{')]
    result=records[-1] if records else {}
    if result.get('undone')!=1:raise ValueError('no_review_decision_to_undo')
    queue=False
    try:queue=revert_queue_identity(lang,card,variant,result.get('candidateIds') or [])

          'humanRouteRemoved':result.get('routeRemoved')==1,'queueUpdated':queue,
          'mappingCacheUpdated':cache_updated,'existingPricesUntouched':True})

def update_case(data):
    lang=data.get("language")
    card=data.get("cardId")
    variant=data.get("variantId")
    status=data.get("status")
    pid=str(data.get("productId") or "")
    note=str(data.get("note") or "").strip()
    acknowledged=data.get("exactPrintConfirmed") is True
    if lang not in ALLOWED_LANG or status not in ALLOWED_STATUS or not card or not variant:
        raise ValueError("invalid identity or status")
    if len(card)>128 or len(variant)>128 or len(note)>500:
        raise ValueError("input length")
    if status=="CONFIRMED_PRINT" and (not acknowledged or not re.fullmatch("[0-9]{3,12}",pid)):
        raise ValueError("confirm exact print and select matching product")
    if status!="CONFIRMED_PRINT":pid=""
    # DB WHERE only allows candidate IDs belonging to checked set; no guide price writes.
    # Single SQL statement: the human-confirmed card/print identity is kept
    # in the review table AND assigned to source routing for future scans.
    # Never write a price here. Preserve a different already-verified provider.
    sql=f"""WITH reviewed AS (
      UPDATE cardmarket_print_identity_review SET
        status={quote(status)}, selected_product_id={quote(pid) if pid else 'NULL'},
        reviewer_note={quote(note)},reviewed_at=NOW()
      WHERE language_code={quote(lang)} AND card_id={quote(card)}
        AND variant_id={quote(variant)} AND status IN ('PENDING_REVIEW','DEFERRED')
        AND ({quote(status)}<>'CONFIRMED_PRINT' OR
          (matched_set_ids ? {quote(pid)} AND candidate_ids ? {quote(pid)}))
      RETURNING language_code,card_id,variant_id,selected_product_id
    ), mapped AS (
      INSERT INTO price_source_routing
        (language_code,card_id,variant_id,primary_provider,provider_product_id,
         source_evidence,verification_state,fallback_provider,refresh_days,next_attempt_at)
      SELECT language_code,card_id,variant_id,'CARDMARKET',selected_product_id,
        'Human confirmed exact set, number, language, edition, foil and stamp',
        'VERIFIED_PRODUCT_ID','TCGDEX',7,NOW()
      FROM reviewed WHERE selected_product_id IS NOT NULL
      ON CONFLICT(language_code,card_id,variant_id) DO UPDATE SET
        primary_provider='CARDMARKET',
        provider_product_id=EXCLUDED.provider_product_id,
        provider_product_url=CASE
          WHEN price_source_routing.provider_product_id=EXCLUDED.provider_product_id
           AND price_source_routing.primary_provider='CARDMARKET'
          THEN price_source_routing.provider_product_url ELSE NULL END,
        source_evidence=EXCLUDED.source_evidence,
        verification_state='VERIFIED_PRODUCT_ID',
        next_attempt_at=NOW(),updated_at=NOW()
      WHERE price_source_routing.verification_state<>'VERIFIED_PRODUCT_ID'
        OR (price_source_routing.primary_provider='CARDMARKET'
            AND price_source_routing.provider_product_id=EXCLUDED.provider_product_id)
      RETURNING card_id
    )
    SELECT (SELECT COUNT(*) FROM reviewed)::int || ':' ||
           (SELECT COUNT(*) FROM mapped)::int;"""
    result=pg(sql)
    if not result.startswith('1:'):raise ValueError("case not found, already reviewed, or product not matched to set")
    routed=result.endswith(':1') if status=='CONFIRMED_PRINT' else False
    if status=='CONFIRMED_PRINT' and not routed:
        # A different verified primary price source is deliberately preserved.
        emit({"ok":True,"cardId":card,"status":status,
              "sourceRoutingUpdated":False,"routingNote":"Existing verified provider preserved"})
        return
    cache_updated=refresh_print_mapping_cache() if status=='CONFIRMED_PRINT' else False
    queue_updated=False
    if routed:
        try:queue_updated=sync_queue_identity(lang,card,variant,pid)
        except (OSError,ValueError):pass  # DB confirmation remains durable; next queue export repairs CSV.
    emit({"ok":True,"cardId":card,"status":status,
          "sourceRoutingUpdated":routed,"scanQueueUpdated":queue_updated,
          "mappingCacheUpdated":cache_updated})
def main():
    a=argparse.ArgumentParser()
    a.add_argument("--action",choices=["list","review","undo"],required=True)
    opt=a.parse_args()
    try:
        data=json.load(sys.stdin)
        if not isinstance(data,dict):raise ValueError("payload")
        if opt.action=="list":list_cases(data)
        elif opt.action=="undo":undo_case(data)
        else:update_case(data)
    except (ValueError,TypeError,json.JSONDecodeError) as e:
        emit({"error":str(e)[:250]});sys.exit(2)
    except (OSError,RuntimeError) as e:
        print(str(e)[:500],file=sys.stderr);sys.exit(3)
if __name__=="__main__":main()

def classify(status,entry=None,queue=None):
    if status=='PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED' and entry and queue:
        if (entry.get('catalog_direct_id_verified') is True and
            str(entry.get('product_id') or '')==str(queue.get('catalog_singleton_id') or '') and
            str(queue.get('catalog_expansion_id') or '').isdigit()):
            # Only unfilterable special prints need a human. A standard
            # normal/holo/reverse can be handled by the existing filtered
            # price pipeline; identity evidence by itself is never a price.
            attrs=filter_readiness.analyze(queue)
            if attrs['unsupported'] or attrs['missing'] or str(queue.get('sonderdruck') or '').strip():
                return 'DRUCKPRUEFUNG'
            return 'PREISPRUEFUNG'
    # A separately known and set-guarded card page makes same-name search
    # failures redundant: the only missing information is the physical stamp
    # or foil. Keep these in a DISTINCT print-review tab; do not falsely
    # suggest the user must identify which Pokémon or set it is again.
    if queue and status in ('MULTIPLE_SEARCH_HITS','NO_EXACT_NUMBER_SEARCH_HIT','PRODUCT_ID_NOT_VERIFIED_FOR_CARD'):
        ref=str(queue.get('reference_product_url') or '')
        stamp=str(queue.get('stempel') or '[]')
        foil=str(queue.get('foil') or '')
        sid=str(queue.get('set_id') or '')
        guard_path=Path('/home/tv/cardshop/config/cardmarket-set-url-guards.json')
        try:guards=json.loads(guard_path.read_text(encoding='utf-8'))
        except (OSError,ValueError):guards={}
        try:
            parsed=urlsplit(ref)
            approved=parsed.scheme=='https' and parsed.hostname=='www.cardmarket.com' and f'/Pokemon/Products/Singles/{guards.get(sid,"__UNKNOWN__")}/' in parsed.path
        except ValueError:approved=False
        if approved and (stamp not in ('','[]') or foil):return 'DRUCKPRUEFUNG'
    # Card+product match is established. Incomplete price pagination does NOT
    # create an identity/print decision for the founder. Scope this rule to
    # 30th sets until other releases have been independently audited.
    if status in PRICE_LIMIT_STATUSES and entry and queue and str(queue.get('set_id') or '').startswith('30th'):
        pid=str(entry.get('product_id') or '')
        candidates={x.strip() for x in str(queue.get('cardmarket_ids') or '').split(',') if x.strip()}
        known_route=str(queue.get('known_url') or '')
        if entry.get('identity_confirmed') is True and pid.isdigit() and pid in candidates and (known_route or queue.get('routing_state')=='VERIFIED_PRODUCT_ID'):
            return 'PREISPRUEFUNG'
        if entry.get('identity_confirmed') is True and pid.isdigit() and candidates and pid not in candidates:
            return 'ZUORDNUNG'

    if status=='MISSING_REQUIRED_SCAN_FIELDS':return 'DATEN_FEHLEN'
    if status in SUCCESS:return None
    if status in TECHNICAL or status.startswith('DEFERRED') or status.startswith('FILTER_') or status.startswith('NO_CONFIRMED'):return 'TECHNISCH'
    if status.startswith('COMPLETE_') or status in ('STARTED','RUNNING'):return None
    return 'ZUORDNUNG'

def reviewed_decisions(language):
    """Current decisions for exact (card, physical variant), DB read only."""
    if language not in ('de','en'):raise ValueError('invalid_language')
    sql=("SELECT card_id,variant_id,status,reviewed_at::text,COALESCE(selected_product_id,'') "
         "FROM cardmarket_print_identity_review WHERE language_code='"+language+
         "' AND status IN ('CONFIRMED_PRINT','NO_MATCH','DEFERRED') AND reviewed_at IS NOT NULL")
    try:
        p=subprocess.run(['docker','exec','cardyssey-db','psql','-X','-qAt','-v','ON_ERROR_STOP=1','-F','|','-U','cardyssey','-d','cardyssey','-c',sql],
                         capture_output=True,text=True,check=True,timeout=12)
    except (subprocess.SubprocessError,OSError):return {}
    out={}
    for line in p.stdout.splitlines():
        fields=line.split('|')
        if len(fields)==5:out[(fields[0],fields[1])]={'status':fields[2],'reviewedAt':fields[3],'selectedProductId':fields[4]}
    return out

def resolved_after_scan(observation,decision):
    if not decision or decision.get('status') not in ('CONFIRMED_PRINT','NO_MATCH','DEFERRED'):return False
    # Once the exact physical print has been human-confirmed, a subsequent
    # name-only/no-result browser failure does not erase that confirmation.
    # A NEW positive contradictory product ID is still surfaced for debugging.
    if decision['status']=='CONFIRMED_PRINT':
        product=str(observation.get('product_id') or '')
        chosen=str(decision.get('selectedProductId') or '')
        if chosen and (not product or product==chosen):return True
    try:
        reviewed=datetime.fromisoformat(str(decision['reviewedAt']).replace('Z','+00:00'))
        seen=datetime.fromisoformat(str(observation.get('seen_at') or '').replace('Z','+00:00'))
        if reviewed.tzinfo is None:reviewed=reviewed.replace(tzinfo=timezone.utc)
        if seen.tzinfo is None:seen=seen.replace(tzinfo=timezone.utc)
        return reviewed>=seen
    except (TypeError,ValueError):return False

def list_failures(base=BASE,language='de',page=0,search='',scope='current',kind='karten',decisions=None):

        except (ValueError,KeyError):pass

#!/usr/bin/env python3
"""Median of ALL eligible Cardmarket asking offers, not the cheapest subset.

Fail closed on incomplete pages or print/language mismatches. One listing counts
once regardless of its quantity; quantity-weighted median is diagnostic only.
Never writes collection/price cache.
"""
import argparse
import json
import re
import statistics
from collections import Counter
from decimal import Decimal, InvalidOperation
from pathlib import Path

VALID_LANGUAGE={"de","en"}
VALID_CONDITION={"NM","MT","NEAR MINT","MINT"}
def money(value):
    if isinstance(value,(int,float,Decimal)):
        v=Decimal(str(value))
    elif isinstance(value,str):
        s=value.strip().replace("€","").replace(" ","")
        if not re.fullmatch(r"[0-9.,]+",s):raise ValueError("invalid price text")
        if "." in s and "," in s:
            s=s.replace(".","").replace(",",".") if s.rfind(",")>s.rfind(".") else s.replace(",","")
        elif "," in s:s=s.replace(",",".")
        elif re.fullmatch(r"\d{1,3}(?:\.\d{3})+",s):s=s.replace(".","")
        v=Decimal(s)
    else:raise ValueError("invalid price type")
    if not v.is_finite() or v<=0:raise ValueError("nonpositive price")
    return v
def median_eur(values):
    return str(statistics.median(values).quantize(Decimal("0.01")))
def evaluate(data):
    expected=data.get("expected") or {}
    lang=expected.get("language")
    pid=str(expected.get("product_id") or "")
    if lang not in VALID_LANGUAGE or not pid.isdigit():
        raise ValueError("expected product ID and de/en required")
    finish=expected.get("finish")
    edition=expected.get("edition")
    if finish not in ("normal","holo","reverse","cosmos"):
        raise ValueError("explicit finish required")
    if edition not in ("unlimited","first_edition","other"):
        raise ValueError("explicit edition required")
    rows=data.get("offers")
    if not isinstance(rows,list):raise ValueError("offers list required")
    prices=[];weighted=[];seen=set();reject=Counter()
    for row in rows:
        if not isinstance(row,dict):reject["invalid_row"]+=1;continue
        id=str(row.get("offer_id") or "").strip()
        if not id or id in seen:reject["duplicate_or_missing_id"]+=1;continue
        seen.add(id)
        if str(row.get("product_id"))!=pid:reject["product"]+=1;continue
        if row.get("language")!=lang:reject["language"]+=1;continue
        if str(row.get("condition") or "").upper() not in VALID_CONDITION:reject["condition"]+=1;continue
        if row.get("finish")!=finish:reject["finish"]+=1;continue
        if row.get("edition")!=edition:reject["edition"]+=1;continue
        if row.get("signed") is not False or row.get("altered") is not False or row.get("graded") is not False:
            reject["modified_or_unverified"]+=1;continue
        if row.get("bundle") is not False:reject["bundle_or_unverified"]+=1;continue
        try:
            price=money(row.get("price_eur"))
            qty=row.get("quantity")
            if type(qty) is not int or qty<1:raise ValueError("missing quantity")
        except (InvalidOperation,ValueError,TypeError):
            reject["price_or_quantity"]+=1;continue
        prices.append(price)
        if len(weighted)+qty<=100000:weighted.extend([price]*qty)
    pages=data.get("pagination") or {}
    count=pages.get("total_offers")
    complete=(
        pages.get("all_pages_fetched") is True and pages.get("capped") is False
        and type(count) is int and count==len(seen)
        and reject["duplicate_or_missing_id"]==0
    )
    result={"product_id":pid,"language":lang,"finish":finish,"edition":edition,
        "listed_offers":len(seen),"accepted_offers":len(prices),
        "reject_reasons":dict(reject),
        "median_all_offers_eur":median_eur(prices) if prices else None,
        "median_quantity_weighted_eur":median_eur(weighted) if weighted else None,
        "all_pages_complete":complete,
        "status":"COMPLETE_ASKING_MEDIAN" if prices and complete else
            "PARTIAL_ASKING_MEDIAN" if prices else "NO_MATCHING_OFFERS",
        "price_basis":"ACTIVE_ASKING_NOT_SOLD",
        "can_publish_print_price":bool(prices and complete and not reject and
           data.get("print_identity_independently_verified") is True)}
    return result
def main():
    p=argparse.ArgumentParser()
    p.add_argument("input",type=Path)
    p.add_argument("--output",type=Path)
    a=p.parse_args()
    result=evaluate(json.loads(a.input.read_text(encoding="utf-8-sig")))
    out=json.dumps(result,ensure_ascii=False,indent=2)+"\n"
    if a.output:a.output.write_text(out,encoding="utf-8")
    print(out,end="")
if __name__=="__main__":main()

#!/usr/bin/env python3
"""Reorder *existing* Cardmarket recovery queues without losing a single variant.

Priority:
  0  open metacard candidates requiring a specific print check
  1  independently human-confirmed special-print routes
  2  manually queued special prints
  3  other failed special prints
  4  other special prints
  5  manually requested non-special failures
  6  remaining technical / mapping errors
  7  ordinary non-bulk (current-epoch incomplete)
  8  bulk (has feed fallback; not deleted)
The scanner applies its own due/epoch gate, so this only changes order, not
whether a card should be scanned. No price/cache/identity DB writes.
"""
import argparse,csv,fcntl,json,os,re,subprocess,tempfile
from collections import Counter
from datetime import datetime,timezone
from pathlib import Path

BASE=Path('/mnt/c/Users/TV/AppData/Local/Cardyssey/CardmarketCollector')
FAILED={'MULTIPLE_SEARCH_HITS','NO_EXACT_NUMBER_SEARCH_HIT','PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED',
'PRODUCT_ID_NOT_VERIFIED_FOR_CARD','PRODUCT_ID_NOT_IN_VERIFIED_SET','PRODUCT_SET_SLUG_MISMATCH',
'NO_PRODUCT_ID','SET_MAPPING_PENDING','FILTERED_SAMPLE_REVIEW_NOT_MARKET_WIDE',

'PROVIDER_ERROR','PROVIDER_TIMEOUT','CONNECTION_LOST','MISSING_REQUIRED_SCAN_FIELDS'}
def moment(value):
    try:
        d=datetime.fromisoformat(str(value).replace('Z','+00:00'))
        return d if d.tzinfo else d.replace(tzinfo=timezone.utc)
    except (ValueError,TypeError):return None
def is_special(row):
    stamp=str(row.get('stempel') or '').strip()
    return bool(str(row.get('foil') or '').strip() or
        str(row.get('sonderdruck') or '').strip() or
        stamp not in ('','[]','null') or
        str(row.get('drucktyp') or '').lower().strip() not in ('normal','reverse'))
def is_pending_manually(row, obs):
    request=moment(row.get('manual_scan_requested_at'))
    previous=moment(obs.get('seen_at')) if obs else None
    return bool(request and (not previous or request>previous))
def classify(row, obs, known_key=False):
    special=is_special(row)
    manual=is_pending_manually(row,obs)
    failed=bool(obs and str(obs.get('status') or '') in FAILED)
    verified=str(row.get('routing_state') or '')=='VERIFIED_PRODUCT_ID'
    if known_key:return 0
    if verified and special:return 1
    if special and manual:return 2
    if special and failed:return 3
    if special:return 4
    if manual:return 5
    if failed:return 6
    if str(row.get('scan_tier') or '')!='4':return 7
    return 8

def prioritize(rows:list,latest:dict,manual_candidates:set)->tuple[list,dict]:
    items=[]
    for i,row in enumerate(rows):
        key=(row['cardyssey_karte'],row['variante_id'])
        rank=classify(row,latest.get(key),key in manual_candidates)
        items.append((rank,i,row))
    items.sort(key=lambda x:(x[0],x[1]))
    output=[x[2] for x in items]
    if len(output)!=len(rows) or len({(x['cardyssey_karte'],x['variante_id']) for x in output})!=len(rows):
        raise ValueError('Queue rows lost/duplicated')
    return output,{'tiers':dict(sorted(Counter(x[0] for x in items).items())),
      'first15':[{'cardId':r['cardyssey_karte'],'print':str(r.get('foil') or r.get('stempel') or r.get('drucktyp') or ''),
                  'tier':n} for n,_,r in items[:15]]}

def load_latest(path):
    latest={}
    if not path.is_file():return latest
    with path.open(encoding='utf-8-sig') as f:
        for line in f:
            try:
                a=json.loads(line)
                if a.get('card_id') and a.get('variant_id') and a.get('status'):
                    latest[(a['card_id'],a['variant_id'])]=a
            except (json.JSONDecodeError,KeyError,TypeError):continue
    return latest
def pending_candidates():
    cmd=['docker','exec','cardyssey-db','psql','-X','-qAt','-U','cardyssey','-d','cardyssey',
      '-v','ON_ERROR_STOP=1','-c',
      """SELECT COALESCE(json_agg(DISTINCT jsonb_build_object(
      'language',language_code,'cardId',card_id,'variantId',variant_id)),'[]'::json)::text
      FROM cardmarket_variant_mappings WHERE mapping_status='PENDING_REVIEW'"""]
    proc=subprocess.run(cmd,capture_output=True,text=True,timeout=15)
    if proc.returncode:raise RuntimeError(proc.stderr[-350:])
    return {(r['language'],r['cardId'],r['variantId']) for r in json.loads(proc.stdout.strip() or '[]')}
def run(base=BASE,apply=False,candidates=None,recheck_open=False,now=None):
    if recheck_open and not apply:raise ValueError('recheck_open requires apply')
    candidates=pending_candidates() if candidates is None else candidates
    stamp=now or datetime.now(timezone.utc)
    sentinel=base/'open-print-recheck-once-20261111.json'
    mark_rechecks=bool(recheck_open and not sentinel.exists())
    report={'mode':'APPLY' if apply else 'DRY_RUN','selectedOpenMappingCases':len(candidates),
       'recheckOpenRequested':recheck_open,'recheckAlreadyScheduled':bool(recheck_open and sentinel.exists()),
       'languages':{}}
    for lang in ('de','en'):
        path=base/f'queue-{lang}.csv'
        with (base/f'queue-{lang}.review.lock').open('a') as lock:
            fcntl.flock(lock,fcntl.LOCK_EX)
            with path.open(encoding='utf-8-sig',newline='') as f:
                reader=csv.DictReader(f);fields=list(reader.fieldnames or []);rows=list(reader)
            latest=load_latest(base/f'browser-observations-{lang}.jsonl')
            specific={(card,var) for language,card,var in candidates if language==lang}
            marked=0
            # New special-print interpretation warrants exactly ONE retry
            # for cases already seen this epoch, without touching successes.
            if mark_rechecks:
                for row in rows:
                    key=(row['cardyssey_karte'],row['variante_id'])
                    previous=latest.get(key)
                    if key not in specific or not previous or previous.get('status') not in FAILED:continue
                    seen=moment(previous.get('seen_at'))
                    requested=moment(row.get('manual_scan_requested_at'))
                    if seen and (not requested or requested<=seen):
                        row['manual_scan_requested_at']=stamp.isoformat()
                        marked+=1
            out,stats=prioritize(rows,latest,specific)
            report['languages'][lang]={'total':len(rows),
               'openMappingCasesInQueue':sum((a['cardyssey_karte'],a['variante_id']) in specific for a in rows),
               'specialRecheckMarked':marked,**stats}
            if apply:
                fd,temp=tempfile.mkstemp(prefix=f'cardmarket-priority-{lang}-',suffix='.csv',dir=base)
                try:
                    with os.fdopen(fd,'w',newline='',encoding='utf-8-sig') as f:
                        writer=csv.DictWriter(f,fieldnames=fields);writer.writeheader();writer.writerows(out)
                    os.replace(temp,path)
                finally:
                    if os.path.exists(temp):os.unlink(temp)

#!/usr/bin/env python3
"""One-time non-destructive full-epoch recovery of failed or partial Cardmarket cards.

The existing full-rescan epoch already queues historical and never-scanned rows.
Only current-epoch failures need manual_scan_requested_at to become due again.
Do not requeue complete asking medians or catalog-only bulk mapping.
"""
from __future__ import annotations
import argparse,csv,fcntl,json,os,tempfile
from collections import Counter
from datetime import datetime,timezone
from pathlib import Path
BASE=Path('/mnt/c/Users/TV/AppData/Local/Cardyssey/CardmarketCollector')
REPORT=Path('/home/tv/cardshop/reports/cardmarket-review')
COMPLETED={
 'MAPPED_NO_PRICE_REQUIRED','COMPLETE_FILTERED_ASKING_MEDIAN_PRINT_UNVERIFIED',
 'COMPLETE_FILTERED_ASKING_MEDIAN','SOURCE_ONLY','SUCCESS',
}
# Explicitly include incomplete listings, multiple search hits, unresolved
# products, missing-number hits, and transient provider/browser failures.
RETRYABLE={
 'MULTIPLE_SEARCH_HITS','NO_EXACT_NUMBER_SEARCH_HIT',
 'PRODUCT_IN_VERIFIED_SET_PRINT_UNVERIFIED','PRODUCT_ID_NOT_VERIFIED_FOR_CARD',
 'PRODUCT_ID_NOT_IN_VERIFIED_SET','PRODUCT_SET_SLUG_MISMATCH','NO_PRODUCT_ID',
 'SET_MAPPING_PENDING','FILTERED_SAMPLE_REVIEW_NOT_MARKET_WIDE',
 'NO_CONFIRMED_FILTERED_OFFERS','FILTER_UNAVAILABLE','FILTER_UNVERIFIED_EMPTY',
 'FILTER_NOT_APPLIED','DEFERRED_PROVIDER_TIMEOUT','DEFERRED_SOCKET',
 'DEFERRED_SEARCH_RESULTS_UNVERIFIED','BLOCKED','LOGIN_REQUIRED','ERROR',
 'PROVIDER_ERROR','PROVIDER_TIMEOUT','CONNECTION_LOST',
 'MISSING_REQUIRED_SCAN_FIELDS',
}
def parse(value):
    try:
        d=datetime.fromisoformat(str(value).replace('Z','+00:00'))
        return d if d.tzinfo else d.replace(tzinfo=timezone.utc)
    except (TypeError,ValueError):return None

def audit_and_schedule(base=BASE, apply=False, languages=('de','en'), now=None):
    epoch_data=json.loads((base/'full-rescan-epoch.json').read_text(encoding='utf-8-sig'))
    epoch=parse(epoch_data['requestedAt'])
    if not epoch:raise ValueError('invalid epoch')
    stamp=now or datetime.now(timezone.utc)
    if stamp<=epoch:raise ValueError('invalid scheduling time')
    report={'existingEpoch':epoch.isoformat(),'requestedAt':stamp.isoformat(),
            'mode':'SCHEDULED' if apply else 'DRY_RUN','languages':{}}
    for lang in languages:
        if lang not in ('de','en'):raise ValueError('Invalid language')
        latest={}
        with (base/f'browser-observations-{lang}.jsonl').open(encoding='utf-8-sig') as f:
            for line in f:
                try:
                    x=json.loads(line);key=(x['card_id'],x['variant_id'])
                    if x.get('status'):latest[key]=x
                except (TypeError,ValueError,KeyError):continue
        queue=base/f'queue-{lang}.csv'
        counts=Counter();retries=Counter();sample=[];to_mark=set()
        # Lock prevents conflicting bulk retry writers; atomically replace
        # queue so a running scanner continues with its already-loaded batch.
        with (base/f'queue-{lang}.review.lock').open('a') as lock:
            fcntl.flock(lock,fcntl.LOCK_EX)
            with queue.open(encoding='utf-8-sig',newline='') as f:
                reader=csv.DictReader(f);fields=list(reader.fieldnames or []);rows=list(reader)
            if 'manual_scan_requested_at' not in fields:
                raise ValueError('Existing queue missing manual_scan_requested_at')
            for row in rows:
                key=(row['cardyssey_karte'],row['variante_id'])
                obs=latest.get(key)
                if not obs:
                    counts['never_scanned_already_due']+=1;continue
                seen=parse(obs.get('seen_at'))
                if not seen or seen<epoch:
                    counts['historical_only_already_due']+=1;continue
                status=str(obs['status'])
                if status in COMPLETED:
                    counts['completed_untouched']+=1;continue
                if status in RETRYABLE:
                    retries[status]+=1;to_mark.add(key)
                    if len(sample)<20:sample.append({'cardId':key[0],'variantId':key[1],'status':status})
                else:counts['unknown_status_for_manual_review']+=1
            if apply and to_mark:
                for row in rows:
                    if (row['cardyssey_karte'],row['variante_id']) in to_mark:
                        previous=parse(row.get('manual_scan_requested_at'))
                        if previous and previous>stamp:continue
                        row['manual_scan_requested_at']=stamp.isoformat()
                fd,tmp=tempfile.mkstemp(prefix=f'recovery-{lang}-',suffix='.csv',dir=base)
                try:
                    with os.fdopen(fd,'w',encoding='utf-8-sig',newline='') as f:
                        writer=csv.DictWriter(f,fieldnames=fields);writer.writeheader();writer.writerows(rows)
                    os.replace(tmp,queue)
                finally:
                    if os.path.exists(tmp):os.unlink(tmp)
        report['languages'][lang]={'totalQueue':len(rows),
           'requeuedCurrentFailures':len(to_mark),
           'failureStatuses':dict(retries),
           'unscannedAlreadyDue':counts['never_scanned_already_due'],
           'historicalAlreadyDue':counts['historical_only_already_due'],
           'completedUnchanged':counts['completed_untouched'],
           'unknownStatusesForReview':counts['unknown_status_for_manual_review'],
           'sample':sample}
    if apply:
        REPORT.mkdir(parents=True,exist_ok=True)
        path=REPORT/'all-unfinished-recovery-20261110.json'
        path.write_text(json.dumps(report,indent=2,ensure_ascii=False)+'\n',encoding='utf-8')
        report['reportFile']=str(path)
    return report

if __name__=='__main__':
    p=argparse.ArgumentParser()
    p.add_argument('--apply',action='store_true')
    p.add_argument('--languages',choices=['de','en','both'],default='both')
    a=p.parse_args()
    result=audit_and_schedule(apply=a.apply,
            languages=('de','en') if a.languages=='both' else (a.languages,))
    print(json.dumps(result,ensure_ascii=False))

#!/usr/bin/env python3
"""Add browser observations to admin review cases without treating samples as prices."""
import argparse
import csv
import hashlib
import json
import math
import subprocess
from pathlib import Path

VALID = {"FILTERED_SAMPLE_REVIEW_NOT_MARKET_WIDE", "NO_CONFIRMED_FILTERED_OFFERS",
         "COMPLETE_FILTERED_ASKING_MEDIAN_PRINT_UNVERIFIED"}

def select_observations(records, worklist):
    allowed = {(r["cardyssey_karte"], r["variante_id"]):
               set(filter(None, (r.get("cardmarket_ids") or "").split(",")))
               for r in worklist}
    refresh_days = {(r["cardyssey_karte"],r["variante_id"]):
                    (7 if str(r.get("scan_tier")) in ("0","1") else
                     14 if str(r.get("scan_tier")) == "2" else 30)
                    for r in worklist}
    prepared = {}
    for r in records:
        key = (str(r.get("card_id") or ""), str(r.get("variant_id") or ""))
        pid = str(r.get("product_id") or "")
        if r.get("status") not in VALID or pid not in allowed.get(key, set()):
            continue
        sample = r.get("sample_median_eur")
        if sample is not None and (not isinstance(sample, (float,int)) or not math.isfinite(sample) or sample <= 0):
            continue
        count = r.get("verified_lang_nm_offers", r.get("de_nm_offers"))
        lang = r.get("language", "de")
        if lang not in ("de", "en") or type(count) is not int or count < 0 or count > 300:
            continue
        complete = r.get("all_pages_complete") is True
        if r.get("status") == "COMPLETE_FILTERED_ASKING_MEDIAN_PRINT_UNVERIFIED" and not complete:
            continue
        if r.get("status") != "COMPLETE_FILTERED_ASKING_MEDIAN_PRINT_UNVERIFIED":
            complete = False
        prepared[key] = {
            "language_code": lang, "card_id": key[0], "variant_id": key[1],
            "refresh_days":refresh_days[key],
            "query": str(r.get("query") or "")[:255],
            "candidate": {"provider": "CARDMARKET", "product_id": pid,
                          "match_type": "CARDMARKET_FILTERED_ASKING_SAMPLE",
                          "language": lang, "min_condition": "NM",
                          "sample_count": count, "sample_median_eur": sample,
                          "all_pages_complete": complete,
                          "sample_not_market_wide": not complete,
                          "access_mode": str(r.get("access_mode") or "UNKNOWN"),
                          "exact_print_verified": False,
                          "no_verified_print_identity": True,
                          "url": str(r.get("url") or "")[:500]}
        }
    return list(prepared.values())

def sql_for(rows):
    serialized = json.dumps(rows, ensure_ascii=False, separators=(",", ":")).replace("'", "''")
    return """
WITH incoming AS (
 SELECT jsonb_array_elements('""" + serialized + """'::jsonb) AS data
), prepared AS (
 SELECT data->>'language_code' AS language_code,
 data->>'card_id' AS card_id, data->>'variant_id' AS variant_id,
 data->>'query' AS search_query, data->'candidate' AS candidate
 FROM incoming
)
INSERT INTO price_review_cases(language_code,card_id,variant_id,search_status,search_query,search_steps,candidates,failure_reason,checked_at)
SELECT language_code,card_id,variant_id,'CARDMARKET_BROWSER_REVIEW',search_query,
 '["CARDMARKET_BROWSER","LANGUAGE_NM_FILTER"]'::jsonb,
 jsonb_build_array(candidate),
 'Filtered active asking offers only; exact printing/edition/foil/stamp not independently confirmed',NOW()
FROM prepared
ON CONFLICT(language_code,card_id,variant_id) DO UPDATE SET
 candidates=(SELECT COALESCE(jsonb_agg(c.value), '[]'::jsonb)
             FROM jsonb_array_elements(price_review_cases.candidates) c(value)
             WHERE c.value->>'match_type'<>'CARDMARKET_FILTERED_ASKING_SAMPLE')
           || EXCLUDED.candidates,
 checked_at=NOW();
"""

def main():
    p = argparse.ArgumentParser()
    p.add_argument("--results", type=Path, required=True)
    p.add_argument("--queue", type=Path, required=True)
    p.add_argument("--apply", action="store_true")

    if args.apply and state_path.exists() and state_path.read_text().strip() == fingerprint:
        print("browser_results_unchanged")
        return
    records = []
    for line in source_bytes.decode("utf-8-sig").splitlines():
        try:
            records.append(json.loads(line))

            continue
    ready = select_observations(records, queue)

                      "validated_review_candidates":len(ready), "apply":args.apply}))

        return
    for i in range(0, len(ready), 100):

                       text=True, stdout=subprocess.DEVNULL)
    # Retain the stable product ID and canonical URL for later revisit.
    # SOURCE_ONLY never implies an authenticated, print-verified price.
    for i in range(0,len(ready),100):
        batch=ready[i:i+100]
        payload=json.dumps(batch,ensure_ascii=False,separators=(",", ":")).replace("'", "''")
        routing_sql="""

        data->>'card_id' AS card_id, data->>'variant_id' AS variant_id,

INSERT INTO price_source_routing

SELECT language_code,card_id,variant_id,'CARDMARKET',product_id,
       product_url,'Browser filtered asking offers; exact print unverified',

FROM fields
WHERE product_url LIKE ('https://www.cardmarket.com/' || language_code || '/Pokemon/Products/Singles/%')
ON CONFLICT(language_code,card_id,variant_id) DO UPDATE SET
 provider_product_id=CASE
   WHEN price_source_routing.verification_state='VERIFIED_PRODUCT_ID'

 provider_product_url=CASE

   ELSE EXCLUDED.provider_product_url END,
 refresh_days=EXCLUDED.refresh_days,
 next_attempt_at=EXCLUDED.next_attempt_at,
 updated_at=NOW()
WHERE price_source_routing.verification_state<>'VERIFIED_PRODUCT_ID';
"""
        subprocess.run(["docker","exec","-i","cardyssey-db","psql", "-v","ON_ERROR_STOP=1",
                        "-U","cardyssey","-d","cardyssey","-q"],input=routing_sql,
                       text=True,check=True,stdout=subprocess.DEVNULL)
    state_path.write_text(fingerprint + "\n", encoding="utf-8")
    print("review_cases_synced", len(ready))

if __name__ == "__main__":
    main()

    else:

def explicit_special_product(name, variant):
    text=str(name or "").lower()
    foil=str(variant.get("foil") or "").lower().replace("_","-")

            scores.append((overlap,overlap/max(1,len(names)),eid))

    if not scores:

    top=scores[0]
    second=scores[1] if len(scores)>1 else (0,0,None)

def valid_link(url):
    if not isinstance(url,str) or len(url)>750:return False
    try:
        p=urlsplit(url)
        if p.scheme!='https' or p.netloc!='www.cardmarket.com' or p.fragment or p.username or p.password:return False
        if re.fullmatch(r'/(de|en)/Pokemon/Products/Singles/[a-zA-Z0-9\-_.~%/]+',p.path):return True
        if p.path=='/Pokemon/Products' and re.fullmatch(r'[0-9]{3,12}',parse_qs(p.query).get('idProduct',[''])[0]):return True
    except ValueError:pass
    return False

def catalog(feed):
    cache=Path('/home/tv/.local/state/cardyssey-queue/price-feeds')/(hashlib.sha256(feed.encode()).hexdigest()+'.json')
    data=json.loads(cache.read_text(encoding='utf-8'))
    return {str(x['idProduct']):x for x in data.get('products' if feed==PRODUCT_URL else 'priceGuides',[])}

def db(sql):
    p=subprocess.run(['docker','exec','cardyssey-db','psql','-X','-qAt','-v','ON_ERROR_STOP=1','-U','cardyssey','-d','cardyssey','-c',sql],capture_output=True,text=True)
    if p.returncode:raise RuntimeError(p.stderr[-450:])
    return p.stdout.strip()

def quote(s):return "'"+str(s).replace("'","''")+"'"

def record(lang,card,variant):
    if lang not in ('de','en') or not re.fullmatch(r'[a-zA-Z0-9_-]{1,128}',card or '') or not re.fullmatch(r'[a-zA-Z0-9_.:-]{1,128}',variant or ''):

     selected_product_id,reviewer_note FROM cardmarket_print_identity_review
     WHERE language_code={quote(lang)} AND card_id={quote(card)} AND variant_id={quote(variant)}
     ) r"""
    text=db(sql)
    if not text:raise ValueError('review_case_not_found')
    return json.loads(text)

function Send-Cdp([string]$method, [hashtable]$parameters) {
  $script:id += 1
  $currentId = $script:id
  $packet = @{id=$currentId;method=$method;params=$parameters} | ConvertTo-Json -Depth 15 -Compress
  $bytes = [text.encoding]::UTF8.GetBytes($packet)
  $sendCts=[threading.CancellationTokenSource]::new([TimeSpan]::FromSeconds(20))
  try {
    $ws.SendAsync([ArraySegment[byte]]::new($bytes),[Net.WebSockets.WebSocketMessageType]::Text,$true,$sendCts.Token).GetAwaiter().GetResult() | Out-Null
  } finally { $sendCts.Dispose() }
  $buffer = [byte[]]::new(262144)
  $all = New-Object System.IO.MemoryStream
  for ($i=0; $i -lt 200; $i++) {
    $cts=[threading.CancellationTokenSource]::new([TimeSpan]::FromSeconds(20))
    try {
      $response = $ws.ReceiveAsync([ArraySegment[byte]]::new($buffer),$cts.Token).GetAwaiter().GetResult()
    } finally { $cts.Dispose() }
    $all.Write($buffer,0,$response.Count)
    if (!$response.EndOfMessage) {continue}
    $raw = [text.encoding]::UTF8.GetString($all.ToArray())
    $all.SetLength(0)
    $parsed = $raw | ConvertFrom-Json
    if ($parsed.id -ne $currentId) {continue}
    if ($parsed.error) { throw "Browser protocol error: $($parsed.error.message)" }
    return $parsed
  }
  throw "Browser protocol timed out"
}
function Js([string]$expression) {
  $reply = Send-Cdp "Runtime.evaluate" @{expression=$expression;returnByValue=$true;awaitPromise=$true}
  if ($reply.result.exceptionDetails) {throw "Browser script exception"}
  return $reply.result.result.value
}
function Move-To([string]$url) {
  Send-Cdp "Page.navigate" @{url=$url} | Out-Null
  $desired = [uri]$url
  for ($i=0;$i -lt 38;$i++) {
    Start-Sleep -Milliseconds 600
    $state=Js 'JSON.stringify({path:location.pathname,href:location.href,ready:document.readyState,body:document.body?.innerText.length||0})'
    if (!$state) {continue}
    $s=$state | ConvertFrom-Json
    $actual=[uri]$s.href
    $expectedQuery=[uri]::UnescapeDataString($desired.Query.TrimStart('?').Replace('+',' '))
    $actualQuery=[uri]::UnescapeDataString($actual.Query.TrimStart('?').Replace('+',' '))
    $sameQuery=($expectedQuery -eq $actualQuery) -or ($desired.AbsolutePath -ne '/de/Pokemon/Products/Search')
    if ($s.path -eq $desired.AbsolutePath -and $sameQuery -and $s.ready -eq "complete" -and $s.body -gt 300) {return $true}
  }
  return $false
}
